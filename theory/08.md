3.2 Proving the Data Race

```cpp
#include <thread>
#include <iostream>
#include <vector>
#include <atomic>

// ===== DEMONSTRATING THE DATA RACE =====
/*
 * Run this and you'll see counter is NOT 2,000,000
 * It will be some random number less than 2,000,000
 * And it changes EVERY run (non-deterministic!)
 */
long long unsafeCounter = 0;

void unsafeIncrement(int times) {
    for (int i = 0; i < times; i++) {
        // THIS IS NOT ATOMIC!
        // The CPU might interrupt between any of these steps:
        // 1. Read unsafeCounter into register
        // 2. Add 1 to register  
        // 3. Write register back to unsafeCounter
        unsafeCounter++;
    }
}

void proveDataRace() {
    const int ITERATIONS = 1'000'000;
    
    // Run 10 experiments
    for (int experiment = 0; experiment < 5; experiment++) {
        unsafeCounter = 0;
        
        std::vector<std::thread> threads;
        for (int i = 0; i < 4; i++) {
            threads.emplace_back(unsafeIncrement, ITERATIONS);
        }
        for (auto& t : threads) t.join();
        
        long long expected = 4LL * ITERATIONS;
        std::cout << "Expected: " << expected 
                  << " Got: " << unsafeCounter
                  << " Lost: " << (expected - unsafeCounter) << "\n";
    }
    /*
     * Expected: 4000000 Got: 2847392 Lost: 1152608
     * Expected: 4000000 Got: 3102847 Lost: 897153
     * Expected: 4000000 Got: 2991827 Lost: 1008173
     * ...
     * Different every time!
     */
}

// ===== HOW THE CPU MAKES IT WORSE: REORDERING =====
/*
 * Modern CPUs and compilers REORDER instructions for performance!
 * This is valid in single-threaded code but causes chaos in multi-threaded.
 *
 * Example:
 * int a = 1;
 * int b = 2;
 * int c = a + b;
 *
 * CPU might execute as:
 * int b = 2;  ← reordered!
 * int a = 1;
 * int c = a + b;
 *
 * In single-threaded: doesn't matter
 * In multi-threaded: Thread 2 might see b=2 but a=0 (not yet written!)
 *
 * THIS IS WHY WE NEED MEMORY ORDERING GUARANTEES
 * Mutexes and atomics provide these guarantees
 */

// ===== FALSE SHARING — The Sneaky Performance Killer =====
/*
 * Even when threads access DIFFERENT variables, they can interfere!
 * 
 * CPU cache works in "cache lines" — typically 64 bytes
 * If two variables are on the SAME cache line:
 * - Thread 1 modifies variable A
 * - This invalidates the ENTIRE cache line
 * - Thread 2 must reload its variable B (even though it didn't change!)
 * - This "cache ping-pong" destroys performance
 */

struct NaiveCounters {
    int c1 = 0; // ← on same cache line!
    int c2 = 0; // ← same cache line → false sharing!
    int c3 = 0;
    int c4 = 0;
};

struct PaddedCounters {
    alignas(64) int c1 = 0; // ← forced to separate cache line
    alignas(64) int c2 = 0;
    alignas(64) int c3 = 0;
    alignas(64) int c4 = 0;
};

/*
 * PaddedCounters version can be 10x faster in multi-threaded code!
 * Each thread accesses its own cache line → no conflicts
 */
```
