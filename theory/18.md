7.3 Spinlock — When Locks Are Too Heavy

```cpp
#include <atomic>
#include <thread>

/*
 * SPINLOCK vs MUTEX:
 * 
 * std::mutex:
 * - If locked, thread SUSPENDS (OS puts it to sleep)
 * - Wake-up is expensive: OS must reschedule (microseconds)
 * - Good when critical section is LONG (sleeping is worthwhile)
 * - Good when many threads contending (sleeping prevents wasting CPU)
 * 
 * Spinlock:
 * - If locked, thread SPINS (busy-loops, checks repeatedly)
 * - No OS involvement, no context switch
 * - Good when critical section is VERY SHORT (nanoseconds)
 * - Good when contention is LOW
 * - Bad if critical section is long (wastes CPU spinning)
 * 
 * RULE OF THUMB:
 * - Critical section < ~50ns: Spinlock
 * - Critical section > ~50ns: Mutex
 */

class SpinLock {
    /*
     * std::atomic_flag:
     * The ONLY atomic type GUARANTEED to be lock-free on all platforms.
     * Can only be in two states: SET or CLEAR.
     * 
     * test_and_set(): sets to SET, returns previous value
     * clear(): sets to CLEAR
     */
    std::atomic_flag flag_ = ATOMIC_FLAG_INIT; // starts CLEAR
    
public:
    void lock() {
        // "I want the lock. Keep trying until I get it."
        while (flag_.test_and_set(std::memory_order_acquire)) {
            /*
             * flag_.test_and_set() returns:
             * - false: flag was CLEAR (unlocked), now SET. I got the lock!
             * - true: flag was already SET (locked). Still locked. Keep spinning.
             * 
             * OPTIMIZATION: TTAS (Test-Test-And-Set)
             * Instead of repeatedly calling test_and_set (which writes),
             * first do a cheap READ. Only try to write when it looks unlocked.
             * This reduces cache coherency traffic.
             */
            #if defined(__cpp_lib_atomic_flag_test) // C++20
            while (flag_.test(std::memory_order_relaxed)) {
                // Just read (cheap) until it looks unlocked
                std::this_thread::yield(); // hint: let other thread run
            }
            // Now try to actually acquire
            #endif
        }
    }
    
    bool try_lock() {
        return !flag_.test_and_set(std::memory_order_acquire);
    }
    
    void unlock() {
        flag_.clear(std::memory_order_release);
        // Clear means "unlocked" — other spinners will see this
    }
};

// ===== SPINLOCK WITH EXPONENTIAL BACKOFF =====
/*
 * Problem with simple spinlock: under high contention,
 * many threads spin simultaneously, thrashing the cache.
 * 
 * Solution: when spinning fails, wait increasingly longer
 * before retrying. Reduces bus contention.
 */
class BackoffSpinLock {
    std::atomic_flag flag_ = ATOMIC_FLAG_INIT;
    
    void cpuRelax() {
        // Tell CPU "I'm spinning, reduce power consumption"
        // On x86: PAUSE instruction
        // On ARM: YIELD instruction
        #if defined(__x86_64__) || defined(__i386__)
        __builtin_ia32_pause();
        #else
        std::this_thread::yield();
        #endif
    }
    
public:
    void lock() {
        int backoff = 1;
        const int MAX_BACKOFF = 1024;
        
        while (flag_.test_and_set(std::memory_order_acquire)) {
            for (int i = 0; i < backoff; i++) {
                cpuRelax(); // pause briefly
            }
            backoff = std::min(backoff * 2, MAX_BACKOFF);
        }
    }
    
    void unlock() {
        flag_.clear(std::memory_order_release);
    }
};
```
