9.2 Bounded Queue — Complete Implementation with Explanation

```cpp
#include <queue>
#include <mutex>
#include <condition_variable>
#include <optional>
#include <stdexcept>
#include <chrono>

/*
 * WHAT "BOUNDED" MEANS AND WHY IT MATTERS:
 * 
 * An unbounded queue can grow without limit.
 * If producer is faster than consumer:
 *   Queue grows → uses more memory → eventually OOM crash!
 * 
 * A bounded queue has a maximum size.
 * If producer tries to add when queue is FULL:
 *   Producer BLOCKS (waits) until consumer removes an item.
 * This is called BACKPRESSURE — pressure flows backwards!
 * 
 * BACKPRESSURE is crucial for system stability:
 * - Prevents memory exhaustion
 * - Automatically slows down fast producers
 * - Creates natural flow control
 * 
 * REAL WORLD ANALOGY:
 * A loading dock with limited parking spots.
 * If all spots full, trucks must wait before unloading.
 * This prevents the warehouse from being overwhelmed!
 */

template<typename T>
class BoundedStreamQueue {
    std::queue<T> queue_;
    const size_t maxSize_;
    
    mutable std::mutex mutex_;
    std::condition_variable notEmpty_; // "hey consumers, item available"
    std::condition_variable notFull_;  // "hey producers, space available"
    
    bool closed_ = false;
    bool noMoreInput_ = false; // producers done, but consumers can drain
    
    // Statistics
    std::atomic<size_t> totalPushed_{0};
    std::atomic<size_t> totalPopped_{0};
    std::atomic<size_t> totalBlocked_{0};
    
public:
    explicit BoundedStreamQueue(size_t maxSize) : maxSize_(maxSize) {
        if (maxSize == 0) throw std::invalid_argument("Size must be > 0");
    }
    
    // ===================================================================
    // PUSH: Add item (blocks if full)
    // ===================================================================
    enum class PushResult { SUCCESS, QUEUE_CLOSED, QUEUE_FULL };
    
    PushResult push(T item) {
        std::unique_lock<std::mutex> lock(mutex_);
        
        if (closed_) return PushResult::QUEUE_CLOSED;
        
        // Block if full (this IS the backpressure mechanism!)
        if (queue_.size() >= maxSize_) {
            totalBlocked_++;
            notFull_.wait(lock, [this]() {
                return queue_.size() < maxSize_ || closed_;
            });
        }
        
        if (closed_) return PushResult::QUEUE_CLOSED;
        
        queue_.push(std::move(item));
        totalPushed_++;
        
        // Wake ONE consumer (not all — spurious wakeups are expensive)
        notEmpty_.notify_one();
        
        return PushResult::SUCCESS;
    }
    
    // Non-blocking push: returns immediately if full
    PushResult try_push(T item) {
        std::unique_lock<std::mutex> lock(mutex_);
        
        if (closed_) return PushResult::QUEUE_CLOSED;
        if (queue_.size() >= maxSize_) return PushResult::QUEUE_FULL;
        
        queue_.push(std::move(item));
        totalPushed_++;
        notEmpty_.notify_one();
        
        return PushResult::SUCCESS;
    }
    
    // Push with timeout
    PushResult push_timed(T item, std::chrono::milliseconds timeout) {
        std::unique_lock<std::mutex> lock(mutex_);
        
        if (closed_) return PushResult::QUEUE_CLOSED;
        
        if (queue_.size() >= maxSize_) {
            bool success = notFull_.wait_for(lock, timeout, [this]() {
                return queue_.size() < maxSize_ || closed_;
            });
            if (!success) return PushResult::QUEUE_FULL;
            if (closed_) return PushResult::QUEUE_CLOSED;
        }
        
        queue_.push(std::move(item));
        totalPushed_++;
        notEmpty_.notify_one();
        return PushResult::SUCCESS;
    }
    
    // ===================================================================
    // POP: Remove item (blocks if empty)
    // ===================================================================
    std::optional<T> pop() {
        std::unique_lock<std::mutex> lock(mutex_);
        
        // Wait until: has item OR (closed AND empty) OR no more input AND empty
        notEmpty_.wait(lock, [this]() {
            return !queue_.empty() || closed_ || noMoreInput_;
        });
        
        if (queue_.empty()) return std::nullopt;
        
        T item = std::move(queue_.front());
        queue_.pop();
        totalPopped_++;
        
        // Wake ONE producer (space available)
        notFull_.notify_one();
        
        return item;
    }
    
    // Non-blocking pop
    std::optional<T> try_pop() {
        std::unique_lock<std::mutex> lock(mutex_);
        
        if (queue_.empty()) return std::nullopt;
        
        T item = std::move(queue_.front());
        queue_.pop();
        totalPopped_++;
        notFull_.notify_one();
        return item;
    }
    
    // ===================================================================
    // BATCH OPERATIONS (more efficient — one lock for many items)
    // ===================================================================
    size_t push_batch(std::vector<T>& items) {
        std::unique_lock<std::mutex> lock(mutex_);
        if (closed_) return 0;
        
        size_t pushed = 0;
        for (auto& item : items) {
            if (queue_.size() >= maxSize_) break; // stop if full
            queue_.push(std::move(item));
            pushed++;
        }
        totalPushed_ += pushed;
        
        if (pushed > 0) {
            // Notify multiple consumers if we pushed multiple items
            if (pushed == 1) notEmpty_.notify_one();
            else notEmpty_.notify_all();
        }
        return pushed;
    }
    
    std::vector<T> pop_batch(size_t maxItems) {
        std::unique_lock<std::mutex> lock(mutex_);
        
        notEmpty_.wait(lock, [this]() {
            return !queue_.empty() || closed_ || noMoreInput_;
        });
        
        std::vector<T> result;
        size_t count = std::min(maxItems, queue_.size());
        result.reserve(count);
        
        for (size_t i = 0; i < count; i++) {
            result.push_back(std::move(queue_.front()));
            queue_.pop();
        }
        
        totalPopped_ += result.size();
        
        if (!result.empty()) {
            notFull_.notify_all(); // freed space, wake producers
        }
        return result;
    }
    
    // ===================================================================
    // LIFECYCLE MANAGEMENT
    // ===================================================================
    
    // Producers done: signal consumers to drain and exit
    void producersDone() {
        std::lock_guard<std::mutex> lock(mutex_);
        noMoreInput_ = true;
        notEmpty_.notify_all(); // wake sleeping consumers
    }
    
    // Hard shutdown: drop all items, wake everyone
    void close() {
        {
            std::lock_guard<std::mutex> lock(mutex_);
            closed_ = true;
            // Optionally clear the queue:
            // while (!queue_.empty()) queue_.pop();
        }
        notEmpty_.notify_all();
        notFull_.notify_all();
    }
    
    // ===================================================================
    // INSPECTION (for monitoring/debugging)
    // ===================================================================
    size_t size() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return queue_.size();
    }
    
    bool empty() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return queue_.empty();
    }
    
    bool full() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return queue_.size() >= maxSize_;
    }
    
    float utilization() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return (float)queue_.size() / maxSize_;
    }
    
    struct Stats {
        size_t totalPushed;
        size_t totalPopped;
        size_t totalBlocked;
        size_t currentSize;
    };
    
    Stats getStats() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return {
            totalPushed_.load(),
            totalPopped_.load(),
            totalBlocked_.load(),
            queue_.size()
        };
    }
};

// ===================================================================
// DEMONSTRATION: Log Processing Pipeline
// ===================================================================
struct LogEntry {
    std::string timestamp;
    std::string level;
    std::string message;
};

struct ProcessedLog {
    LogEntry original;
    bool isError;
    std::string summary;
};

void log_processing_demo() {
    // Bounded queues at each stage
    BoundedStreamQueue<LogEntry> rawLogs(1000);
    BoundedStreamQueue<ProcessedLog> processedLogs(500);
    
    // Stage 1: Log generator (simulates reading from file/network)
    std::thread logGenerator([&rawLogs]() {
        for (int i = 0; i < 10000; i++) {
            rawLogs.push(LogEntry{
                "2024-01-01T" + std::to_string(i),
                i % 10 == 0 ? "ERROR" : "INFO",
                "Message " + std::to_string(i)
            });
        }
        rawLogs.producersDone(); // Signal: no more logs coming
    });
    
    // Stage 2: Log processors (multiple threads process in parallel)
    std::vector<std::thread> processors;
    for (int i = 0; i < 4; i++) {
        processors.emplace_back([&rawLogs, &processedLogs, i]() {
            while (auto log = rawLogs.pop()) {
                // Process the log (simulate work)
                std::this_thread::sleep_for(std::chrono::microseconds(100));
                
                bool isError = (log->level == "ERROR");
                processedLogs.push(ProcessedLog{
                    std::move(*log),
                    isError,
                    isError ? "ERROR DETECTED" : "OK"
                });
            }
            // This processor is done — but queue might still have items
            // Other processors still running
        });
    }
    
    // Stage 3: Wait for all processors, then signal consumer
    std::thread coordinatorThread([&processors, &processedLogs]() {
        for (auto& p : processors) p.join();
        processedLogs.producersDone();
    });
    
    // Stage 4: Output consumer
    std::thread outputConsumer([&processedLogs]() {
        int errorCount = 0;
        while (auto processed = processedLogs.pop()) {
            if (processed->isError) errorCount++;
        }
        std::cout << "Processing complete. Errors found: " << errorCount << "\n";
    });
    
    logGenerator.join();
    coordinatorThread.join();
    outputConsumer.join();
}
```
