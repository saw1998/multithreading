7.2 Atomics — Complete Guide

```cpp
#include <atomic>
#include <thread>
#include <vector>
#include <iostream>
#include <cassert>

// ===================================================================
// std::atomic<T> — Thread-safe variable without locks
// ===================================================================
/*
 * std::atomic<T> works for:
 * - All integral types (int, long, etc.) — lock-free on most platforms
 * - Pointers — lock-free
 * - bool — lock-free
 * - Custom types (but may fall back to internal lock!)
 * 
 * Use atomic<T>::is_lock_free() or is_always_lock_free to check.
 */

// ===== BASIC ATOMICS =====
void basic_atomics() {
    std::atomic<int> x{0};
    
    // ===== LOAD =====
    /*
     * NOT safe:  int val = x;  (may be non-atomic on some platforms)
     * SAFE:      int val = x.load();
     * 
     * INTERNALLY: Uses hardware load instruction with memory barrier
     */
    int val = x.load(std::memory_order_seq_cst);
    
    // ===== STORE =====
    /*
     * NOT safe:  x = 42;
     * SAFE:      x.store(42);
     */
    x.store(42, std::memory_order_seq_cst);
    
    // ===== EXCHANGE =====
    /*
     * Atomically: reads current value, writes new value, returns OLD value
     * 
     * INTERNALLY (pseudocode):
     * int exchange(int newVal) {
     *   LOCK_BUS;
     *   int old = *this;
     *   *this = newVal;
     *   UNLOCK_BUS;
     *   return old;
     * }
     */
    int old = x.exchange(100); // x becomes 100, old = 42
    
    // ===== FETCH_ADD =====
    /*
     * Atomically: reads, adds, returns OLD value
     * 
     * THIS is what makes the counter thread-safe!
     * Unlike x++, this is one atomic instruction
     */
    int before = x.fetch_add(5); // x becomes 105, before = 100
    int after = x.load();        // after = 105
    
    // Also: fetch_sub, fetch_and, fetch_or, fetch_xor
    
    // ===== COMPARE AND SWAP (CAS) — THE FUNDAMENTAL PRIMITIVE =====
    /*
     * This is the MOST IMPORTANT atomic operation.
     * Almost all lock-free algorithms use CAS.
     * 
     * SEMANTICS:
     * if (x == expected) {
     *   x = desired;
     *   return true;
     * } else {
     *   expected = x;  // update expected with current value
     *   return false;
     * }
     * 
     * CRUCIAL: This entire if-else is ONE ATOMIC OPERATION!
     * No thread can interfere between the comparison and the swap.
     * 
     * HARDWARE: Maps to CMPXCHG instruction on x86
     * The CPU locks the cache line, does compare+swap, releases.
     */
    
    // compare_exchange_strong: guarantees it only fails if expected ≠ x
    int expected = 105;
    bool success = x.compare_exchange_strong(expected, 999);
    // If x was 105: x becomes 999, success=true, expected unchanged
    // If x was not 105: x unchanged, success=false, expected=current x value
    
    std::cout << "CAS success: " << success 
              << " x: " << x.load() 
              << " expected: " << expected << "\n";
    
    // compare_exchange_weak: MAY spuriously fail (returns false even if equal)
    // Use in loops where retrying is fine — can be faster on some architectures
    expected = 999;
    while (!x.compare_exchange_weak(expected, 777)) {
        // expected is now current value of x
        // loop until we succeed
    }
}

// ===================================================================
// MEMORY ORDERING — The Most Complex Part
// ===================================================================
/*
 * THE PROBLEM:
 * 
 * Modern CPUs and compilers REORDER instructions for performance.
 * This is invisible in single-threaded code but breaks multi-threading.
 * 
 * Example:
 * 
 * int data = 0;
 * std::atomic<bool> ready{false};
 * 
 * Thread 1:
 *   data = 42;        // instruction A
 *   ready = true;     // instruction B
 * 
 * Thread 2:
 *   while (!ready);   // instruction C
 *   assert(data == 42); // instruction D — MIGHT FAIL!
 * 
 * CPU might reorder Thread 1's instructions as: B, A
 * Thread 2 might see ready=true but data=0!
 * 
 * Memory orders specify ordering guarantees between operations.
 */

std::atomic<bool> dataReady{false};
int sharedData = 0;

// ===== MEMORY_ORDER_RELAXED =====
/*
 * Guarantees: Only atomicity of this one operation.
 * Does NOT guarantee ordering relative to other operations.
 * 
 * USE WHEN: 
 * - Just counting (each increment is atomic, order doesn't matter)
 * - Statistics collection
 * - Setting flags where order doesn't matter
 * 
 * FASTEST — no memory barriers inserted
 */
std::atomic<int> hitCount{0};
void recordHit() {
    hitCount.fetch_add(1, std::memory_order_relaxed);
    // We don't care if other reads/writes are reordered around this
}

// ===== MEMORY_ORDER_RELEASE + MEMORY_ORDER_ACQUIRE =====
/*
 * RELEASE semantics (on stores):
 * "Everything I did BEFORE this store is visible to anyone
 *  who ACQUIRES the same atomic variable."
 * Prevents: earlier writes from being reordered AFTER the release store.
 * 
 * ACQUIRE semantics (on loads):
 * "I can see everything done before the matching RELEASE store."
 * Prevents: later reads from being reordered BEFORE the acquire load.
 * 
 * TOGETHER: Creates a "happens-before" relationship:
 * Everything before release-store happens-before everything
 * after acquire-load of the same atomic variable.
 */

void producer_release() {
    sharedData = 42;  // (1) Write data
    
    // RELEASE store: guarantees (1) is visible before this
    dataReady.store(true, std::memory_order_release);  // (2)
}

void consumer_acquire() {
    // ACQUIRE load: guarantees we see everything before matching release
    while (!dataReady.load(std::memory_order_acquire)) {  // (3)
        std::this_thread::yield();
    }
    
    // GUARANTEED: sharedData == 42
    // Because: (1) happens-before (2) [release ordering]
    //          (2) happens-before (3) [synchronization]
    //          (3) happens-before here [acquire ordering]
    assert(sharedData == 42);  // (4) — always safe
}

// ===== MEMORY_ORDER_SEQ_CST (SEQUENTIAL CONSISTENCY) =====
/*
 * The strongest, simplest, and most intuitive ordering.
 * This is the DEFAULT for all atomic operations.
 * 
 * Guarantee: There is a SINGLE GLOBAL ORDER of all seq_cst operations.
 * Every thread sees ALL seq_cst operations in the SAME order.
 * 
 * Think of it as: all atomic operations happen in a single
 * global timeline that all threads agree on.
 * 
 * COST: Requires full memory barriers — the most expensive.
 * But on x86, stores are already strongly ordered, so seq_cst
 * is close in performance to release/acquire.
 * On ARM/PowerPC, the difference is more significant.
 * 
 * RECOMMENDATION: Use seq_cst (default) unless profiling shows
 * it's a bottleneck. Correctness > micro-optimization.
 */

// ===================================================================
// CAS-BASED ALGORITHMS — Classic Interview Patterns

// ===== LOCK-FREE COUNTER =====
class LockFreeCounter {
    std::atomic<long long> count_{0};
    
public:
    void increment() {
        count_.fetch_add(1, std::memory_order_relaxed);
    }
    
    void decrement() {
        count_.fetch_sub(1, std::memory_order_relaxed);
    }
    
    long long get() const {
        return count_.load(std::memory_order_relaxed);
    }
    
    // Reset and return old value
    long long reset() {
        return count_.exchange(0, std::memory_order_acq_rel);
    }
};

// ===== LOCK-FREE STACK =====
/*
 * This demonstrates the CAS loop pattern used in lock-free algorithms:
 * 
 * 1. Read current state
 * 2. Compute desired new state
 * 3. CAS: atomically set new state IF current state = what we read
 * 4. If CAS fails (someone else changed it), retry from step 1
 * 
 * This is called "optimistic concurrency":
 * Assume no conflict, proceed, if conflict detected, retry.
 */

template<typename T>
class LockFreeStack {
    struct Node {
        T data;
        Node* next;
        
        explicit Node(T val) : data(std::move(val)), next(nullptr) {}
    };
    
    std::atomic<Node*> head_{nullptr};
    
public:
    void push(T value) {
        Node* newNode = new Node(std::move(value));
        
        // CAS loop:
        // "I think head is X. If it still is, make my node the new head.
        //  If someone else changed head, update my assumption and retry."
        newNode->next = head_.load(std::memory_order_relaxed);
        
        while (!head_.compare_exchange_weak(
            newNode->next,   // expected: current head
            newNode,         // desired: my new node  
            std::memory_order_release,   // success ordering
            std::memory_order_relaxed    // failure ordering
        )) {
            // On failure, newNode->next is updated to current head
            // (compare_exchange_weak does this automatically)
            // Loop retries with updated expectation
        }
    }
    
    std::optional<T> pop() {
        Node* oldHead = head_.load(std::memory_order_relaxed);
        
        while (oldHead != nullptr) {
            // Try to make the second node the new head
            if (head_.compare_exchange_weak(
                oldHead,           // expected: current head
                oldHead->next,     // desired: next becomes head
                std::memory_order_acquire,  // if success: I own this memory
                std::memory_order_relaxed
            )) {
                // Success! We own oldHead now
                T value = std::move(oldHead->data);
                delete oldHead;
                return value;
            }
            // Failure: oldHead updated to current head, retry
        }
        
        return std::nullopt; // stack was empty
    }
    
    bool empty() const {
        return head_.load(std::memory_order_relaxed) == nullptr;
    }
    
    ~LockFreeStack() {
        while (!empty()) {
            pop();
        }
    }
};

/*
 * ABA PROBLEM — The Lock-Free Nightmare:
 * 
 * Stack: A → B → C
 * 
 * Thread 1: Reads head = A (wants to pop A)
 * Thread 1: Gets suspended
 * 
 * Thread 2: Pops A (now: B → C)
 * Thread 2: Pops B (now: C)
 * Thread 2: Pushes A back (now: A → C)
 *           BUT this is a different A! (or A was freed and reallocated)
 * 
 * Thread 1: Resumes, sees head is still A ← same value!
 * Thread 1: CAS succeeds! Sets head to oldHead->next
 *           BUT oldHead->next is now C (not B!)
 *           B is LOST, and if A was freed, this is USE-AFTER-FREE!
 * 
 * SOLUTIONS:
 * 1. Tagged pointers (pack version counter with pointer)
 * 2. Hazard pointers (mark pointers before using them)
 * 3. Epoch-based reclamation
 * 4. Use atomic<shared_ptr<>> (C++20)
 * 5. Avoid lock-free (use mutex instead)
 * 
 * For interviews: understand the ABA problem exists and name solutions.
 * Production code: use a proven library (like Intel TBB, Boost.Lockfree)
 */
```
