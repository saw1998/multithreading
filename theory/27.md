11.2 Implementing and Using Semaphores

```cpp
#include <mutex>
#include <condition_variable>
#include <thread>
#include <vector>
#include <iostream>
#include <chrono>

// ===================================================================
// SEMAPHORE IMPLEMENTATION (Pre-C++20)
// ===================================================================
/*
 * INTERNAL STATE:
 * - count: number of available permits
 * 
 * acquire():
 *   Wait until count > 0, then decrement
 *   If count == 0, sleep until someone releases
 * 
 * release():
 *   Increment count
 *   Wake one waiter if any
 * 
 * KEY DIFFERENCE FROM MUTEX:
 * - Mutex: the SAME thread must lock and unlock
 * - Semaphore: thread A can release a permit that thread B acquired!
 * - This makes semaphores useful for thread SIGNALING
 */

class Semaphore {
    int count_;
    const int maxCount_;
    std::mutex mutex_;
    std::condition_variable cv_;
    
public:
    explicit Semaphore(int initialCount, int maxCount = INT_MAX)
        : count_(initialCount), maxCount_(maxCount) {
        if (initialCount < 0) throw std::invalid_argument("Count must be >= 0");
    }
    
    // Take a permit (blocks if none available)
    void acquire(int n = 1) {
        std::unique_lock<std::mutex> lock(mutex_);
        cv_.wait(lock, [this, n]() { return count_ >= n; });
        count_ -= n;
    }
    
    // Try to take permit without blocking
    bool try_acquire(int n = 1) {
        std::lock_guard<std::mutex> lock(mutex_);
        if (count_ >= n) {
            count_ -= n;
            return true;
        }
        return false;
    }
    
    // Try with timeout
    bool try_acquire_for(int n, std::chrono::milliseconds timeout) {
        std::unique_lock<std::mutex> lock(mutex_);
        return cv_.wait_for(lock, timeout, [this, n]() { 
            return count_ >= n; 
        }) && (count_ -= n, true);
    }
    
    // Return permits
    void release(int n = 1) {
        {
            std::lock_guard<std::mutex> lock(mutex_);
            if (count_ + n > maxCount_) {
                throw std::overflow_error("Semaphore count would exceed max");
            }
            count_ += n;
        }
        // Notify waiters (outside lock for efficiency)
        if (n == 1) cv_.notify_one();
        else cv_.notify_all();
    }
    
    int available() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return count_;
    }
};

// ===== RAII SEMAPHORE GUARD =====
class SemaphoreGuard {
    Semaphore& sem_;
public:
    explicit SemaphoreGuard(Semaphore& sem) : sem_(sem) { sem_.acquire(); }
    ~SemaphoreGuard() { sem_.release(); }
    SemaphoreGuard(const SemaphoreGuard&) = delete;
};

// ===================================================================
// USE CASE 1: Connection Pool
// ===================================================================
class DatabaseConnectionPool {
    Semaphore availableConnections_;
    std::vector<std::string> connections_;
    std::mutex connectionsMutex_;
    
public:
    explicit DatabaseConnectionPool(int maxConnections)
        : availableConnections_(maxConnections) {
        
        for (int i = 0; i < maxConnections; i++) {
            connections_.push_back("connection_" + std::to_string(i));
        }
    }
    
    std::string acquireConnection() {
        // Wait for available connection
        availableConnections_.acquire();
        
        std::lock_guard<std::mutex> lock(connectionsMutex_);
        std::string conn = connections_.back();
        connections_.pop_back();
        return conn;
    }
    
    void releaseConnection(std::string conn) {
        {
            std::lock_guard<std::mutex> lock(connectionsMutex_);
            connections_.push_back(std::move(conn));
        }
        availableConnections_.release();
    }
};

// ===================================================================
// USE CASE 2: Thread Signaling (Binary Semaphore)
// ===================================================================
/*
 * Binary semaphore is like a mutex but:
 * - Starts at 0 (no permits)
 * - Thread A releases it (signals)
 * - Thread B acquires it (waits for signal)
 * - A and B can be different threads (mutex can't do this!)
 * 
 * This is one-way signaling: "event happened, you can proceed"
 * 
 * Unlike condition variable:
 * - No spurious wakeups
 * - Signal is "remembered" even if nobody waiting yet
 */

class EventSignal {
    Semaphore sem_{0}; // starts with 0 permits
    
public:
    // Called by signaling thread
    void signal() {
        sem_.release(); // add permit
    }
    
    // Called by waiting thread
    void wait() {
        sem_.acquire(); // wait for permit
    }
    
    bool wait_for(std::chrono::milliseconds timeout) {
        return sem_.try_acquire_for(1, timeout);
    }
};

void signaling_demo() {
    EventSignal startSignal, doneSignal;
    
    std::thread worker([&]() {
        std::cout << "Worker waiting for start signal...\n";
        startSignal.wait(); // blocks here
        
        std::cout << "Worker started! Computing...\n";
        std::this_thread::sleep_for(std::chrono::milliseconds(500));
        std::cout << "Worker done!\n";
        
        doneSignal.signal(); // tell main we're done
    });
    
    std::this_thread::sleep_for(std::chrono::milliseconds(200));
    std::cout << "Main sending start signal\n";
    startSignal.signal(); // unblock worker
    
    std::cout << "Main waiting for worker to finish...\n";
    doneSignal.wait(); // wait for worker's done signal
    std::cout << "Main confirmed worker done!\n";
    
    worker.join();
}

// ===================================================================
// USE CASE 3: Throttling / Rate Limiting
// ===================================================================
/*
 * Allow at most N concurrent operations.
 * When N are running, new requests block.
 * When one finishes, one more can start.
 */

class ConcurrencyLimiter {
    Semaphore semaphore_;
    
public:
    explicit ConcurrencyLimiter(int maxConcurrent)
        : semaphore_(maxConcurrent) {}
    
    template<typename Fn>
    auto run(Fn&& fn) -> decltype(fn()) {
        semaphore_.acquire(); // wait for slot
        
        struct Guard {
            Semaphore& sem;
            ~Guard() { sem.release(); } // always release when done
        } guard{semaphore_};
        
        return fn(); // run with limited concurrency
    }
};

// ===================================================================
// C++20 std::counting_semaphore and std::binary_semaphore
// ===================================================================
#include <semaphore>

void cpp20_semaphore_demo() {
    // Counting semaphore with max value 10
    std::counting_semaphore<10> sem{3}; // starts with 3 permits
    
    sem.acquire();      // blocks if no permits
    sem.release();      // returns a permit
    bool ok = sem.try_acquire(); // non-blocking
    
    // With timeout (C++20):
    bool acquired = sem.try_acquire_for(std::chrono::milliseconds(100));
    (void)acquired;
    
    // Binary semaphore (max 1)
    std::binary_semaphore signal{0}; // starts "unset"
    
    std::thread t([&signal]() {
        std::cout << "Waiting...\n";
        signal.acquire(); // wait
        std::cout << "Got signal!\n";
    });
    
    std::this_thread::sleep_for(std::chrono::milliseconds(200));
    signal.release(); // signal the thread
    
    t.join();
}
```
