9.3 Ring Buffer — Circular Fixed-Size Queue

```cpp
/*
 * WHAT IS A RING BUFFER?
 * 
 * A circular array where head and tail pointers "wrap around".
 * Like a clock face: after position N, you go back to position 0.
 * 
 * WHY USE IT INSTEAD OF std::queue?
 * 1. Fixed memory allocation (no dynamic allocation = no fragmentation)
 * 2. Better cache performance (contiguous memory)
 * 3. Perfect for "overwrite oldest" semantics (logging)
 * 4. Lock-free variants are very efficient
 * 
 * VISUALIZATION:
 * 
 * Array: [A][B][C][D][E][_][_][_]
 *              ^              ^
 *              tail           head
 * 
 * Push F: [A][B][C][D][E][F][_][_]
 *              ^                 ^
 *              tail              head
 * 
 * Pop: returns A
 *        [_][B][C][D][E][F][_][_]
 *                 ^              ^
 *                 tail           head
 * 
 * Push G, H, I (wraps around!):
 *        [I][B][C][D][E][F][G][H]
 *                 ^   ^
 *                 tail head
 * 
 * Push J (FULL! head would meet tail):
 *   If "overwrite oldest" mode: overwrite B
 *   If "backpressure" mode: block until consumer reads
 */

template<typename T>
class RingBuffer {
    std::vector<T> buffer_;
    size_t head_ = 0;  // where NEXT write goes
    size_t tail_ = 0;  // where NEXT read comes from
    size_t count_ = 0; // current number of elements
    size_t capacity_;
    
    mutable std::mutex mutex_;
    std::condition_variable notEmpty_;
    std::condition_variable notFull_;
    bool closed_ = false;
    
public:
    explicit RingBuffer(size_t capacity) 
        : buffer_(capacity), capacity_(capacity) {}
    
    bool push(T item) {
        std::unique_lock<std::mutex> lock(mutex_);
        
        notFull_.wait(lock, [this]() {
            return count_ < capacity_ || closed_;
        });
        
        if (closed_) return false;
        
        buffer_[head_] = std::move(item);
        head_ = (head_ + 1) % capacity_; // wrap around!
        ++count_;
        
        notEmpty_.notify_one();
        return true;
    }
    
    // Overwrite mode: always succeeds, may overwrite oldest
    void push_overwrite(T item) {
        std::lock_guard<std::mutex> lock(mutex_);
        
        if (count_ == capacity_) {
            // Buffer full: advance tail (lose oldest item)
            tail_ = (tail_ + 1) % capacity_;
        } else {
            ++count_;
        }
        
        buffer_[head_] = std::move(item);
        head_ = (head_ + 1) % capacity_;
        
        notEmpty_.notify_one();
    }
    
    std::optional<T> pop() {
        std::unique_lock<std::mutex> lock(mutex_);
        
        notEmpty_.wait(lock, [this]() {
            return count_ > 0 || closed_;
        });
        
        if (count_ == 0) return std::nullopt;
        
        T item = std::move(buffer_[tail_]);
        tail_ = (tail_ + 1) % capacity_; // wrap around!
        --count_;
        
        notFull_.notify_one();
        return item;
    }
    
    // Peek without removing
    std::optional<T> peek() const {
        std::lock_guard<std::mutex> lock(mutex_);
        if (count_ == 0) return std::nullopt;
        return buffer_[tail_];
    }
    
    void close() {
        std::lock_guard<std::mutex> lock(mutex_);
        closed_ = true;
        notEmpty_.notify_all();
        notFull_.notify_all();
    }
    
    size_t size() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return count_;
    }
    
    bool full() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return count_ == capacity_;
    }
    
    bool empty() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return count_ == 0;
    }
};
```
