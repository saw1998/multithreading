```cpp
#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>
#include <functional>
#include <future>
#include <vector>
#include <atomic>
#include <iostream>

/*
 * THREAD POOL: The most important concurrency pattern.
 * 
 * WHY NOT CREATE A THREAD FOR EVERY TASK?
 * 
 * Thread creation is expensive:
 * - OS must allocate stack memory (1-8MB per thread)
 * - OS schedules the thread
 * - Creating 1000 threads for 1000 tasks = slow!
 * 
 * Thread pools REUSE threads:
 * - Create N threads once at startup
 * - Give them tasks from a queue
 * - Threads live until pool is destroyed
 * - Submit 1000 tasks with 8 threads → tasks run N at a time
 * 
 * ARCHITECTURE:
 * ┌────────────────────────────────────────────────────────────┐
 * │                    Thread Pool                            │
 * │                                                           │
 * │  Submit →  [Task Queue]  ← workers pull from             │
 * │            [Task Queue]                                   │
 * │            [Task Queue]                                   │
 * │                                                           │
 * │  Worker 1  Worker 2  Worker 3  ... Worker N               │
 * │  (Thread)  (Thread)  (Thread)      (Thread)               │
 * └────────────────────────────────────────────────────────────┘
 */

class ThreadPool {
    // =========================================================
    // WORKER THREADS
    // =========================================================
    std::vector<std::thread> workers_;
    
    // =========================================================
    // TASK QUEUE + SYNCHRONIZATION
    // =========================================================
    std::queue<std::function<void()>> taskQueue_;
    mutable std::mutex queueMutex_;
    std::condition_variable taskAvailable_;
    
    // =========================================================
    // LIFECYCLE STATE
    // =========================================================
    bool stop_ = false;
    std::atomic<int> activeTasks_{0};
    std::condition_variable allDone_;
    
    // =========================================================
    // INTERNAL: Worker thread function
    // =========================================================
    void workerLoop() {
        while (true) {
            std::function<void()> task;
            
            {
                std::unique_lock<std::mutex> lock(queueMutex_);
                
                // Wait for: a task OR shutdown signal
                taskAvailable_.wait(lock, [this]() {
                    return !taskQueue_.empty() || stop_;
                });
                
                // If shutting down and no more tasks: exit
                if (stop_ && taskQueue_.empty()) return;
                
                // Grab task from queue
                task = std::move(taskQueue_.front());
                taskQueue_.pop();
                activeTasks_++;
            }
            
            // Execute task OUTSIDE the lock!
            // (So other workers can grab tasks while this one executes)
            try {
                task();
            } catch (const std::exception& e) {
                std::cerr << "Task threw exception: " << e.what() << "\n";
            } catch (...) {
                std::cerr << "Task threw unknown exception\n";
            }
            
            {
                std::lock_guard<std::mutex> lock(queueMutex_);
                activeTasks_--;
                if (activeTasks_ == 0 && taskQueue_.empty()) {
                    allDone_.notify_all();
                }
            }
        }
    }
    
public:
    // =========================================================
    // CONSTRUCTOR: Creates worker threads
    // =========================================================
    explicit ThreadPool(size_t numThreads) {
        for (size_t i = 0; i < numThreads; ++i) {
            workers_.emplace_back(&ThreadPool::workerLoop, this);
        }
    }
    
    // =========================================================
    // submit(): Submit a task and get a future for its result
    // =========================================================
    /*
     * Template magic:
     * F = any callable (lambda, function, etc.)
     * Args = argument types
     * std::invoke_result_t<F, Args...> = return type of F(args...)
     * 
     * We wrap the task in a packaged_task to capture the result
     * and make it available via a future.
     */
    template<typename F, typename... Args>
    auto submit(F&& fn, Args&&... args) 
        -> std::future<std::invoke_result_t<F, Args...>> 
    {
        using ReturnType = std::invoke_result_t<F, Args...>;
        
        // Wrap in packaged_task to capture return value
        auto task = std::make_shared<std::packaged_task<ReturnType()>>(
            [fn = std::forward<F>(fn), 
             ...args = std::forward<Args>(args)]() mutable {
                return fn(std::forward<Args>(args)...);
            }
        );
        
        std::future<ReturnType> future = task->get_future();
        
        {
            std::lock_guard<std::mutex> lock(queueMutex_);
            
            if (stop_) {
                throw std::runtime_error("Cannot submit to stopped pool");
            }
            
            // Wrap packaged_task in void() lambda for the queue
            taskQueue_.emplace([task]() { (*task)(); });
        }
        
        taskAvailable_.notify_one(); // wake one worker
        return future;
    }
    
    // =========================================================
    // WAITING AND LIFECYCLE
    // =========================================================
    
    // Wait for all currently submitted tasks to complete
    void waitForAll() {
        std::unique_lock<std::mutex> lock(queueMutex_);
        allDone_.wait(lock, [this]() {
            return taskQueue_.empty() && activeTasks_ == 0;
        });
    }
    
    size_t pendingTasks() const {
        std::lock_guard<std::mutex> lock(queueMutex_);
        return taskQueue_.size();
    }
    
    int activeTasks() const {
        return activeTasks_.load();
    }
    
    size_t threadCount() const {
        return workers_.size();
    }
    
    // =========================================================
    // DESTRUCTOR: Graceful shutdown
    // =========================================================
    ~ThreadPool() {
        {
            std::lock_guard<std::mutex> lock(queueMutex_);
            stop_ = true; // signal all workers to stop
        }
        taskAvailable_.notify_all(); // wake all sleeping workers
        
        for (auto& worker : workers_) {
            worker.join(); // wait for each worker to finish
        }
    }
    
    // Non-copyable, non-moveable
    ThreadPool(const ThreadPool&) = delete;
    ThreadPool& operator=(const ThreadPool&) = delete;
};

// ===================================================================
// USAGE EXAMPLES
// ===================================================================

void thread_pool_demos() {
    ThreadPool pool(std::thread::hardware_concurrency());
    
    // ===== SIMPLE TASK SUBMISSION =====
    auto f1 = pool.submit([]() {
        return 42;
    });
    std::cout << "Result: " << f1.get() << "\n";
    
    // ===== TASKS WITH ARGUMENTS =====
    auto f2 = pool.submit([](int a, int b) {
        return a + b;
    }, 10, 20);
    std::cout << "Sum: " << f2.get() << "\n";
    
    // ===== FIRE AND FORGET =====
    pool.submit([]() {
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        std::cout << "Background task done\n";
    });
    
    // ===== PARALLEL COMPUTATION =====
    const int N = 1000000;
    std::vector<std::future<long long>> futures;
    
    int numChunks = std::thread::hardware_concurrency();
    int chunkSize = N / numChunks;
    
    for (int i = 0; i < numChunks; i++) {
        int start = i * chunkSize;
        int end = (i == numChunks - 1) ? N : start + chunkSize;
        
        futures.push_back(pool.submit([start, end]() {
            long long sum = 0;
            for (int j = start; j < end; j++) sum += j;
            return sum;
        }));
    }
    
    long long total = 0;
    for (auto& f : futures) total += f.get();
    
    std::cout << "Parallel sum 0.." << N << " = " << total << "\n";
    
    // ===== EXCEPTION HANDLING =====
    auto failingTask = pool.submit([]() -> int {
        throw std::runtime_error("Task failed!");
        return 0;
    });
    
    try {
        failingTask.get(); // exception rethrown here
    } catch (const std::exception& e) {
        std::cout << "Caught: " << e.what() << "\n";
    }
    
    pool.waitForAll();
    std::cout << "All tasks complete\n";
}
```
