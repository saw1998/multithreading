```cpp
#include <thread>
#include <iostream>
#include <chrono>
#include <stdexcept>

/*
 * WHY join() exists:
 * When a thread finishes, it becomes a "zombie" — 
 * it's done but its resources aren't cleaned up yet.
 * join() cleans up those resources AND waits for completion.
 *
 * WHY detach() exists:
 * Sometimes you want "fire and forget" — launch background
 * work and never think about it again.
 * detach() separates the thread object from the actual thread.
 * The thread runs until it finishes, then cleans itself up.
 *
 * WHY joinable() exists:
 * A thread is "joinable" if join()/detach() hasn't been called yet.
 * Destroying a joinable thread → std::terminate() is called!
 * This is a safety check.
 */

// ===== LIFECYCLE DEMO =====
void shortTask(int id, int durationMs) {
    std::cout << "[Thread " << id << "] Starting\n";
    std::this_thread::sleep_for(std::chrono::milliseconds(durationMs));
    std::cout << "[Thread " << id << "] Finishing\n";
}

void lifecycle_demo() {
    // === CREATE ===
    std::thread t(shortTask, 1, 500);
    
    std::cout << "Thread created. Joinable: " 
              << t.joinable() << "\n"; // 1 (true)
    std::cout << "Thread ID: " << t.get_id() << "\n";
    
    // === WAITING / BLOCKING ===
    // Meanwhile, main thread can do other work...
    std::cout << "Main thread doing other work...\n";
    
    // === JOIN ===
    t.join(); // Block until thread finishes
    
    std::cout << "After join. Joinable: " 
              << t.joinable() << "\n"; // 0 (false)
    
    // === DETACH ===
    std::thread bg(shortTask, 2, 1000);
    bg.detach(); // "fire and forget"
    std::cout << "Detached. Joinable: " 
              << bg.joinable() << "\n"; // 0 (false)
    
    // bg is now independent — but be careful!
    // If main exits before bg finishes → undefined behavior!
    std::this_thread::sleep_for(std::chrono::milliseconds(1500)); // give bg time
}

// ===== SAFE RAII THREAD WRAPPER =====
/*
 * WHY THIS IS NEEDED:
 * 
 * void dangerousFunction() {
 *     std::thread t(doWork);
 *     riskyOperation(); // ← if this throws, t's destructor is called
 *     // but t is still joinable → std::terminate() called!
 *     t.join();
 * }
 * 
 * RAII (Resource Acquisition Is Initialization) pattern
 * ensures cleanup happens automatically even if exceptions occur.
 */
class ManagedThread {
    std::thread thread_;
    
public:
    // Can specify join or detach behavior
    enum class JoinPolicy { JOIN, DETACH };
    
    JoinPolicy policy_;
    
    template<typename Fn, typename... Args>
    explicit ManagedThread(JoinPolicy policy, Fn&& fn, Args&&... args)
        : thread_(std::forward<Fn>(fn), std::forward<Args>(args)...)
        , policy_(policy) {}
    
    ~ManagedThread() {
        if (thread_.joinable()) {
            if (policy_ == JoinPolicy::JOIN) {
                thread_.join();   // Wait for completion
            } else {
                thread_.detach(); // Let it run free
            }
        }
    }
    
    // Prevent copying (threads are not copyable!)
    ManagedThread(const ManagedThread&) = delete;
    ManagedThread& operator=(const ManagedThread&) = delete;
    
    // Allow moving (transfer ownership)
    ManagedThread(ManagedThread&&) = default;
    ManagedThread& operator=(ManagedThread&&) = default;
    
    std::thread::id get_id() const { return thread_.get_id(); }
    bool joinable() const { return thread_.joinable(); }
};

void safe_usage() {
    try {
        ManagedThread t(ManagedThread::JoinPolicy::JOIN, shortTask, 1, 100);
        
        // Even if this throws, ManagedThread's destructor will join!
        throw std::runtime_error("Something went wrong");
        
        // Without ManagedThread, the std::thread destructor would
        // call std::terminate() here!
    } catch (std::exception& e) {
        std::cout << "Caught: " << e.what() << "\n";
        // ManagedThread's destructor already joined the thread safely
    }
}

// ===== C++20 std::jthread — The "smart" thread =====
/*
 * jthread solves two problems:
 * 1. Auto-joins in destructor (no more manual join/terminate risk)
 * 2. Built-in cooperative cancellation via std::stop_token
 * 
 * Cooperative cancellation means: the thread CHECKS if it should stop.
 * You cannot forcibly kill a thread in C++ (it would leave resources
 * in inconsistent state). The thread must willingly check and exit.
 */
#include <stop_token>

void cancellableWorker(std::stop_token stopToken, int id) {
    int iteration = 0;
    
    // Periodically check if we should stop
    while (!stopToken.stop_requested()) {
        std::cout << "Thread " << id << " iteration " << iteration++ << "\n";
        
        // Interruptible sleep — wakes up if stop is requested!
        std::this_thread::sleep_for(std::chrono::milliseconds(200));
    }
    
    // Cleanup code here — runs before thread exits
    std::cout << "Thread " << id << " shutting down gracefully\n";
}

void jthread_demo() {
    {
        std::jthread worker(cancellableWorker, 1);
        
        std::this_thread::sleep_for(std::chrono::milliseconds(1000));
        
        // Request the thread to stop
        worker.request_stop();
        
        // jthread automatically joins in its destructor!
        // No need to call worker.join()
    } // ← destructor called here: joins automatically
    
    std::cout << "Worker finished\n";
}

// ===== HOW THREADS PASS ARGUMENTS =====
/*
 * This is a common source of bugs!
 * Thread arguments are COPIED by default (like std::bind)
 */
void byValue(int x) { x = 999; } // won't affect original
void byRef(int& x) { x = 999; }  // WILL affect original

void argument_demo() {
    int value = 42;
    
    // COPY: value stays 42
    std::thread t1(byValue, value);
    t1.join();
    std::cout << value << "\n"; // 42
    
    // REFERENCE: value becomes 999
    // MUST use std::ref() to pass by reference!
    // Without std::ref, it would try to copy a reference → compiler error
    std::thread t2(byRef, std::ref(value));
    t2.join();
    std::cout << value << "\n"; // 999
    
    // MOVE: transfer ownership of resource to thread
    std::string str = "hello";
    std::thread t3([](std::string s) {
        std::cout << "Got: " << s << "\n";
    }, std::move(str));  // str is now empty
    t3.join();
    std::cout << "str is now: '" << str << "'\n"; // empty
}
```
