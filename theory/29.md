#include <latch>
#include <barrier>
#include <thread>
#include <vector>
#include <iostream>
#include <chrono>
#include <numeric>

// ===================================================================
// IMPLEMENTING LATCH FROM SCRATCH (to understand internals)
// ===================================================================
/*
 * Latch INTERNAL STATE:
 * - count: atomic counter, initialized to N
 * - When count reaches 0, all waiters are released
 * 
 * count_down(): atomically decrement count
 *               if count reaches 0: wake all waiters
 * 
 * wait(): block until count == 0
 * 
 * arrive_and_wait(): count_down() then wait()
 */

class ManualLatch {
    std::atomic<int> count_;
    std::mutex mutex_;
    std::condition_variable cv_;
    
public:
    explicit ManualLatch(int count) : count_(count) {
        if (count <= 0) throw std::invalid_argument("Count must be positive");
    }
    
    // Signal that one unit of work is done
    void count_down(int n = 1) {
        int remaining = count_.fetch_sub(n) - n;
        
        if (remaining == 0) {
            // Last one! Wake all waiters
            cv_.notify_all();
        } else if (remaining < 0) {
            throw std::logic_error("Latch count went below 0");
        }
    }
    
    // Block until count reaches 0
    void wait() {
        std::unique_lock<std::mutex> lock(mutex_);
        cv_.wait(lock, [this]() { return count_.load() == 0; });
    }
    
    // count_down() + wait() atomically (well, sequentially under lock)
    void arrive_and_wait() {
        count_down();
        wait();
    }
    
    bool try_wait() const {
        return count_.load() == 0;
    }
};

// ===================================================================
// USE CASE 1: Parallel Initialization
// ===================================================================
/*
 * Common pattern: several subsystems must initialize before
 * the application can serve requests.
 * 
 * With latch: start all initialization in parallel,
 * main thread waits for all to complete.
 */

void parallel_initialization_demo() {
    struct InitResult {
        std::string subsystem;
        bool success;
        std::string message;
    };
    
    const int NUM_SUBSYSTEMS = 5;
    std::latch initLatch(NUM_SUBSYSTEMS);
    std::vector<InitResult> results(NUM_SUBSYSTEMS);
    
    auto subsystem = [&](int id, std::chrono::milliseconds initTime) {
        // Simulate initialization
        std::this_thread::sleep_for(initTime);
        
        results[id] = {
            "Subsystem_" + std::to_string(id),
            true,
            "Initialized in " + std::to_string(initTime.count()) + "ms"
        };
        
        std::cout << results[id].subsystem << " ready\n";
        initLatch.count_down(); // Signal: I'm ready
    };
    
    std::vector<std::thread> threads;
    for (int i = 0; i < NUM_SUBSYSTEMS; i++) {
        threads.emplace_back(subsystem, i, 
                             std::chrono::milliseconds(100 + i * 50));
    }
    
    std::cout << "Waiting for all subsystems...\n";
    initLatch.wait(); // Block until all N count_down() calls
    std::cout << "All subsystems ready! Starting server.\n";
    
    for (auto& t : threads) t.join();
}

// ===================================================================
// USE CASE 2: One-Shot Signal (Latch as starting gun)
// ===================================================================
void starting_gun_demo() {
    const int NUM_THREADS = 8;
    std::latch startLatch(1); // count of 1, triggered by one signal
    std::latch doneLatch(NUM_THREADS);
    
    std::vector<std::thread> workers;
    std::vector<std::chrono::nanoseconds> startTimes(NUM_THREADS);
    
    for (int i = 0; i < NUM_THREADS; i++) {
        workers.emplace_back([&, i]() {
            startLatch.wait(); // all threads wait here
            
            // All threads start as close to simultaneously as possible!
            startTimes[i] = std::chrono::high_resolution_clock::now()
                              .time_since_epoch();
            
            // Do work...
            volatile long sum = 0;
            for (int j = 0; j < 1000000; j++) sum += j;
            
            doneLatch.count_down();
        });
    }
    
    std::this_thread::sleep_for(std::chrono::milliseconds(100));
    std::cout << "Ready... Set... GO!\n";
    startLatch.count_down(); // FIRE! All threads start simultaneously
    
    doneLatch.wait(); // wait for all to finish
    
    for (auto& t : workers) t.join();
    
    // Verify they started close to simultaneously
    auto [minT, maxT] = std::minmax_element(startTimes.begin(), startTimes.end());
    auto spread = *maxT - *minT;
    std::cout << "Thread start spread: " 
              << std::chrono::duration_cast<std::chrono::microseconds>(spread).count()
              << " microseconds\n";
}

// ===================================================================
// BARRIER — Reusable Synchronization
// ===================================================================
/*
 * BARRIER INTERNAL STATE:
 * - expectedCount: number of threads (set at construction)
 * - arrivalCount: threads that have called arrive_and_wait() this round
 * - phase: which "generation" of synchronization we're in
 * 
 * When all N threads arrive:
 * 1. Call the completion function (if any)
 * 2. Reset arrivalCount to 0
 * 3. Increment phase
 * 4. Wake all waiting threads
 * 5. Start next "phase"
 * 
 * Threads can then do their next round of work.
 */

class ManualBarrier {
    const int expected_;
    std::atomic<int> arrived_{0};
    std::atomic<int> phase_{0};
    std::mutex mutex_;
    std::condition_variable cv_;
    std::function<void()> completion_;
    
public:
    explicit ManualBarrier(int count, std::function<void()> fn = {})
        : expected_(count), completion_(std::move(fn)) {}
    
    void arrive_and_wait() {
        int myPhase = phase_.load();
        
        int prev = arrived_.fetch_add(1);
        
        if (prev + 1 == expected_) {
            // Last to arrive — do completion, then release everyone
            if (completion_) completion_();
            
            arrived_.store(0);
            phase_.fetch_add(1); // increment phase
            cv_.notify_all();
        } else {
            // Wait for all to arrive
            std::unique_lock<std::mutex> lock(mutex_);
            cv_.wait(lock, [this, myPhase]() {
                return phase_.load() > myPhase;
            });
        }
    }
    
    void arrive_and_drop() {
        // Arrive but don't wait (leave the barrier group)
        // Decrements expected count for future phases
        // (Complex to implement correctly, simplified here)
        arrive_and_wait();
    }
};

// ===================================================================
// USE CASE: Parallel Sorting (Phases Need Synchronization)
// ===================================================================
/*
 * Parallel merge sort requires barriers between phases:
 * Phase 1: All threads sort their chunk
 * BARRIER: Wait for all chunks sorted
 * Phase 2: Merge adjacent chunks pairwise
 * BARRIER: Wait for all merges
 * Phase 3: Merge again
 * ...
 */

void parallel_computation_demo() {
    const int NUM_THREADS = 4;
    const int ARRAY_SIZE = 1000;
    
    std::vector<int> data(ARRAY_SIZE);
    std::iota(data.begin(), data.end(), 1); // fill with 1, 2, 3...
    
    int phaseNum = 0;
    
    std::barrier syncPoint(NUM_THREADS, [&phaseNum]() noexcept {
        std::cout << "=== Phase " << phaseNum++ << " complete ===\n";
    });
    
    std::vector<long long> partialSums(NUM_THREADS, 0);
    
    auto worker = [&](int id) {
        int chunkSize = ARRAY_SIZE / NUM_THREADS;
        int start = id * chunkSize;
        int end = (id == NUM_THREADS - 1) ? ARRAY_SIZE : start + chunkSize;
        
        // Phase 1: Each thread computes sum of its chunk
        for (int i = start; i < end; i++) {
            partialSums[id] += data[i];
        }
        std::cout << "Thread " << id << " partial sum: " 
                  << partialSums[id] << "\n";
        
        syncPoint.arrive_and_wait(); // ← BARRIER: all threads sync here
        
        // Phase 2: Thread 0 combines all partial sums
        if (id == 0) {
            long long total = 0;
            for (auto sum : partialSums) total += sum;
            std::cout << "Total sum: " << total << "\n";
        }
        
        syncPoint.arrive_and_wait(); // ← BARRIER: sync before phase 3
        
        // Phase 3: Each thread does more work based on total
        // (knowing total is ready because we passed the barrier)
        std::cout << "Thread " << id << " starting phase 3\n";
        
        syncPoint.arrive_and_wait(); // ← Final sync
    };
    
    std::vector<std::thread> threads;
    for (int i = 0; i < NUM_THREADS; i++) {
        threads.emplace_back(worker, i);
    }
    for (auto& t : threads) t.join();
}
