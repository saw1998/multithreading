10.2 Future, Promise, Async — Complete Guide

```cpp
#include <future>
#include <thread>
#include <iostream>
#include <vector>
#include <exception>

// ===================================================================
// std::async — Simplest way to run tasks asynchronously
// ===================================================================
/*
 * std::async creates a task and returns a future.
 * Two launch policies:
 * 
 * std::launch::async:
 *   - Guaranteed to run in a NEW THREAD immediately
 *   - If system can't create thread → std::system_error thrown
 * 
 * std::launch::deferred:
 *   - LAZY: doesn't run until future.get() is called
 *   - Runs in the CALLING THREAD when get() is called
 *   - Useful for lazy evaluation
 * 
 * Default (async|deferred):
 *   - Implementation decides (usually deferred on some platforms!)
 *   - AVOID DEFAULT unless you don't care about when it runs
 */

int expensiveComputation(int n) {
    std::cout << "Computing on thread: " 
              << std::this_thread::get_id() << "\n";
    
    // Simulate expensive work
    long long sum = 0;
    for (int i = 0; i < n; i++) sum += i;
    return (int)(sum % 1000000);
}

void async_examples() {
    // ===== BASIC ASYNC =====
    auto future = std::async(std::launch::async, 
                             expensiveComputation, 1000000);
    
    std::cout << "Task submitted. Main thread continuing...\n";
    // Do other work here!
    
    // Block and get result when needed
    int result = future.get(); // waits for completion
    std::cout << "Result: " << result << "\n";
    
    // ===== DEFERRED (LAZY) =====
    auto lazyFuture = std::async(std::launch::deferred, 
                                  expensiveComputation, 500000);
    
    std::cout << "Deferred task created (NOT running yet)\n";
    
    int lazyResult = lazyFuture.get(); // RUNS NOW, in THIS thread
    std::cout << "Lazy result: " << lazyResult << "\n";
    
    // ===== MULTIPLE PARALLEL TASKS =====
    std::vector<std::future<int>> futures;
    for (int i = 0; i < 8; i++) {
        futures.push_back(
            std::async(std::launch::async, expensiveComputation, 100000 * i)
        );
    }
    
    // Collect all results
    int total = 0;
    for (auto& f : futures) {
        total += f.get(); // each get() blocks until that task finishes
    }
    std::cout << "Total: " << total << "\n";
    
    // ===== EXCEPTION HANDLING =====
    auto badFuture = std::async(std::launch::async, []() -> int {
        throw std::runtime_error("Task failed!");
        return 0; // never reached
    });
    
    try {
        int val = badFuture.get(); // exception rethrown here!
        (void)val;
    } catch (const std::exception& e) {
        std::cout << "Caught: " << e.what() << "\n";
        // Exception is transported across thread boundaries!
    }
}

// ===================================================================
// std::promise and std::future — Manual channel
// ===================================================================
/*
 * promise<T>: The WRITE end of the channel
 *   - set_value(T): push a result
 *   - set_exception(): push an exception
 *   - Can only set ONCE!
 * 
 * future<T>: The READ end of the channel
 *   - get(): block until value ready, then return it
 *   - wait(): block until ready (don't read)
 *   - wait_for(): wait with timeout
 *   - Can only call get() ONCE!
 * 
 * INTERNAL: They share a "shared state" object on the heap.
 * Promise writes to it, Future reads from it.
 * Condition variable inside handles the waiting.
 * 
 * ┌──────────┐        ┌────────────────┐        ┌──────────┐
 * │ promise  │──────► │  shared state  │ ◄──────│  future  │
 * │          │ writes │  (heap alloc)  │ reads  │          │
 * └──────────┘        └────────────────┘        └──────────┘
 */

void promise_examples() {
    // ===== BASIC PROMISE/FUTURE =====
    std::promise<int> promise;
    std::future<int> future = promise.get_future();
    
    std::thread worker([p = std::move(promise)]() mutable {
        try {
            // Simulate work
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
            int result = 42;
            p.set_value(result); // fulfil the promise
        } catch (...) {
            p.set_exception(std::current_exception()); // forward exception!
        }
    });
    
    // Main thread does other work...
    
    // When ready to use result:
    try {
        int result = future.get(); // blocks until worker sets value
        std::cout << "Got: " << result << "\n";
    } catch (const std::exception& e) {
        std::cout << "Worker failed: " << e.what() << "\n";
    }
    
    worker.join();
    
    // ===== SHARED FUTURE (multiple consumers) =====
    /*
     * future<T> can only be consumed ONCE.
     * shared_future<T> can be consumed by MULTIPLE threads.
     * Like a future being broadcast to multiple listeners.
     */
    std::promise<std::string> sharedPromise;
    std::shared_future<std::string> sharedFuture = 
        sharedPromise.get_future().share(); // convert to shared
    
    // Multiple threads can wait on same value
    std::vector<std::thread> waiters;
    for (int i = 0; i < 5; i++) {
        waiters.emplace_back([sharedFuture, i]() {
            const std::string& value = sharedFuture.get(); // all get same value
            std::cout << "Waiter " << i << " got: " << value << "\n";
        });
    }
    
    std::this_thread::sleep_for(std::chrono::milliseconds(100));
    sharedPromise.set_value("broadcast message!"); // all waiters wake up
    
    for (auto& w : waiters) w.join();
    
    // ===== PIPELINE WITH PROMISES =====
    // Create a chain of futures for a processing pipeline
    auto stage1 = std::async(std::launch::async, []() {
        return std::string("raw data");
    });
    
    // stage2 depends on stage1's result
    auto stage2 = std::async(std::launch::async, [f = std::move(stage1)]() mutable {
        std::string input = f.get(); // wait for stage1
        return "processed: " + input;
    });
    
    // stage3 depends on stage2's result
    auto stage3 = std::async(std::launch::async, [f = std::move(stage2)]() mutable {
        std::string input = f.get(); // wait for stage2
        return "final: " + input;
    });
    
    std::cout << stage3.get() << "\n";
}

// ===================================================================
// std::packaged_task — Wrapping a callable for async execution
// ===================================================================
/*
 * packaged_task<ReturnType(ArgTypes...)> wraps a callable.
 * Like binding a function to a future/promise pair.
 * 
 * USE CASES:
 * - You have a callable you want to run asynchronously
 * - You want to run it in a thread pool (can't use async directly)
 * - You want to run it on a specific thread
 * 
 * COMPARISON:
 * async: runs immediately, returns future
 * packaged_task: runs when YOU call it, gives future
 */

void packaged_task_examples() {
    // ===== BASIC PACKAGED TASK =====
    std::packaged_task<int(int, int)> task(
        [](int a, int b) { return a + b; }
    );
    
    std::future<int> future = task.get_future();
    
    // Task is NOT running yet!
    // We can pass it to any thread or run it later
    
    std::thread worker(std::move(task), 10, 20);
    
    std::cout << "Result: " << future.get() << "\n"; // 30
    worker.join();
    
    // ===== PACKAGED TASK IN A THREAD POOL CONTEXT =====
    // This is how thread pools work!
    
    std::queue<std::packaged_task<int()>> taskQueue;
    std::mutex queueMutex;
    std::condition_variable queueCV;
    bool done = false;
    
    // "Thread pool" worker
    std::thread poolWorker([&]() {
        while (true) {
            std::packaged_task<int()> task;
            {
                std::unique_lock<std::mutex> lock(queueMutex);
                queueCV.wait(lock, [&]() {
                    return !taskQueue.empty() || done;
                });
                if (done && taskQueue.empty()) break;
                task = std::move(taskQueue.front());
                taskQueue.pop();
            }
            task(); // execute the task!
        }
    });
    
    // Submit tasks to pool
    std::vector<std::future<int>> results;
    for (int i = 0; i < 5; i++) {
        std::packaged_task<int()> t([i]() { return i * i; });
        results.push_back(t.get_future());
        
        {
            std::lock_guard<std::mutex> lock(queueMutex);
            taskQueue.push(std::move(t));
        }
        queueCV.notify_one();
    }
    
    // Collect results
    for (auto& r : results) {
        std::cout << r.get() << " ";
    }
    std::cout << "\n";
    
    {
        std::lock_guard<std::mutex> lock(queueMutex);
        done = true;
    }
    queueCV.notify_one();
    poolWorker.join();
}
```
