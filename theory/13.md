5.2 All Deadlock Solutions

```cpp
#include <mutex>
#include <thread>
#include <iostream>
#include <algorithm>
#include <vector>

// ===== SIMULATING DEADLOCK =====
std::mutex lockA, lockB;

void thread1_deadlocks() {
    std::lock_guard<std::mutex> la(lockA); // grabs A
    std::this_thread::sleep_for(std::chrono::milliseconds(10)); // "do some work"
    std::lock_guard<std::mutex> lb(lockB); // waits for B
    // Thread 2 holds B and waits for A → DEADLOCK!
}

void thread2_deadlocks() {
    std::lock_guard<std::mutex> lb(lockB); // grabs B
    std::this_thread::sleep_for(std::chrono::milliseconds(10));
    std::lock_guard<std::mutex> la(lockA); // waits for A
    // Thread 1 holds A and waits for B → DEADLOCK!
}

// ===================================================================
// SOLUTION 1: CONSISTENT LOCK ORDERING
// ===================================================================
/*
 * Break CIRCULAR WAIT:
 * Establish a global ordering of all mutexes.
 * Always lock in the same order.
 * 
 * Example: always lock A before B, never B before A.
 * 
 * Thread 1: A then B → waits for A
 * Thread 2: A then B → waits for A (not B!)
 * No cycle → no deadlock
 * 
 * When locks have no natural order, use their memory address as order.
 */

void solution1_locking_order() {
    // Both threads lock in same order: A then B
    std::lock_guard<std::mutex> la(lockA);
    std::lock_guard<std::mutex> lb(lockB);
    std::cout << "Both locks acquired safely\n";
}

// For arbitrary mutexes, sort by address:
void safeTransferWithOrdering(std::mutex& m1, std::mutex& m2) {
    // Always lock lower address first
    std::mutex* first = &m1;
    std::mutex* second = &m2;
    
    if (first > second) std::swap(first, second);
    
    std::lock_guard<std::mutex> l1(*first);
    std::lock_guard<std::mutex> l2(*second);
    
    // critical section
}

// ===================================================================
// SOLUTION 2: std::lock / std::scoped_lock
// ===================================================================
/*
 * Break HOLD AND WAIT:
 * Acquire ALL needed locks atomically — never hold one while waiting for another.
 * 
 * std::lock uses a deadlock-free algorithm:
 * Internally uses "try and back off":
 * 
 * try_lock_a:
 *   if !lock(A): goto try_lock_a
 *   if !try_lock(B):
 *     unlock(A)
 *     sleep briefly
 *     goto try_lock_a
 *   // got both!
 */

void solution2_std_lock(std::mutex& m1, std::mutex& m2) {
    // Method A: std::lock + adopt_lock (pre-C++17)
    std::lock(m1, m2); // deadlock-free multi-lock
    std::lock_guard<std::mutex> l1(m1, std::adopt_lock); // take ownership
    std::lock_guard<std::mutex> l2(m2, std::adopt_lock); // take ownership
    // critical section
}

void solution2_scoped_lock(std::mutex& m1, std::mutex& m2) {
    // Method B: scoped_lock (C++17) — PREFERRED
    std::scoped_lock lock(m1, m2); // deadlock-free, auto-unlock
    // critical section
}

// ===================================================================
// SOLUTION 3: TRY-LOCK WITH BACKOFF
// ===================================================================
/*
 * Break HOLD AND WAIT differently:
 * If we can't get all locks, release everything and try again.
 * 
 * Good for when we can't use scoped_lock (e.g., custom lock types)
 */

bool acquireWithBackoff(std::mutex& m1, std::mutex& m2, int maxRetries = 100) {
    int retries = 0;
    
    while (retries < maxRetries) {
        // Try to lock first mutex
        if (m1.try_lock()) {
            // Try to lock second mutex
            if (m2.try_lock()) {
                return true; // got both!
            }
            m1.unlock(); // couldn't get m2, release m1
        }
        
        // Exponential backoff to reduce contention
        std::this_thread::sleep_for(
            std::chrono::microseconds(1 << retries) // 1, 2, 4, 8... μs
        );
        retries++;
    }
    return false; // timeout
}

// ===================================================================
// SOLUTION 4: LOCK HIERARCHY
// ===================================================================
/*
 * Assign a "level" to each mutex.
 * Rule: you can only lock a mutex at a LOWER level than what you hold.
 * 
 * Level 3: database_mutex
 * Level 2: table_mutex  
 * Level 1: row_mutex
 * 
 * Valid:   lock(db) then lock(table) then lock(row)  (3→2→1)
 * Invalid: lock(row) then lock(table) → DETECTED and thrown!
 * 
 * This ENFORCES consistent ordering at runtime.
 */

class HierarchicalMutex {
    std::mutex internal_mutex;
    const unsigned long hierarchy_value;
    unsigned long previous_hierarchy_value;
    
    // Thread-local: each thread has its own current hierarchy level
    static thread_local unsigned long this_thread_hierarchy_value;
    
    void check_for_hierarchy_violation() {
        if (this_thread_hierarchy_value <= hierarchy_value) {
            throw std::logic_error(
                "Hierarchy violation: trying to lock lower-level mutex "
                "while holding higher-level mutex"
            );
        }
    }
    
    void update_hierarchy_value() {
        previous_hierarchy_value = this_thread_hierarchy_value;
        this_thread_hierarchy_value = hierarchy_value;
    }
    
public:
    explicit HierarchicalMutex(unsigned long value) 
        : hierarchy_value(value)
        , previous_hierarchy_value(0) {}
    
    void lock() {
        check_for_hierarchy_violation();
        internal_mutex.lock();
        update_hierarchy_value();
    }
    
    void unlock() {
        if (this_thread_hierarchy_value != hierarchy_value) {
            throw std::logic_error("Unlocking wrong mutex");
        }
        this_thread_hierarchy_value = previous_hierarchy_value;
        internal_mutex.unlock();
    }
    
    bool try_lock() {
        check_for_hierarchy_violation();
        if (!internal_mutex.try_lock()) return false;
        update_hierarchy_value();
        return true;
    }
};

thread_local unsigned long HierarchicalMutex::this_thread_hierarchy_value(ULONG_MAX);

// Usage:
HierarchicalMutex high_level_mutex(10000);
HierarchicalMutex low_level_mutex(5000);

void do_low_level_stuff() {
    std::lock_guard<HierarchicalMutex> low_lock(low_level_mutex);
    // do work
}

void do_high_level_stuff() {
    std::lock_guard<HierarchicalMutex> high_lock(high_level_mutex);
    do_low_level_stuff(); // OK: 10000 → 5000 (decreasing)
}

void do_bad_stuff() {
    std::lock_guard<HierarchicalMutex> low_lock(low_level_mutex);
    // Trying to lock HIGH level while holding LOW level!
    std::lock_guard<HierarchicalMutex> high_lock(high_level_mutex); // THROWS!
}
```
