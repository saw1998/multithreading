// QUESTION 1: Implement Thread-Safe Singleton
// ===================================================================
/*
 * WHY TRICKY: Simple approach has a race condition!
 * 
 * Naive broken version:
 * Singleton* getInstance() {
 *     if (!instance) {          // Thread A checks: null
 *                               // Thread B checks: null
 *         instance = new Singleton(); // Both create!
 *     }
 *     return instance;
 * }
 */

class Singleton {
    static std::unique_ptr<Singleton> instance_;
    static std::once_flag onceFlag_; // std::once_flag is the key!
    
    Singleton() { std::cout << "Singleton created\n"; }
    
public:
    // ===== SOLUTION 1: std::call_once (PREFERRED) =====
    /*
     * std::call_once guarantees: fn is called EXACTLY ONCE
     * even if multiple threads call it simultaneously.
     * Other threads block until the first call completes.
     */
    static Singleton& getInstance() {
        std::call_once(onceFlag_, []() {
            instance_ = std::make_unique<Singleton>();
        });
        return *instance_;
    }
    
    // ===== SOLUTION 2: Static local variable (C++11 guaranteed) =====
    /*
     * C++11 GUARANTEES: If multiple threads try to initialize
     * a static local variable simultaneously, only one succeeds.
     * Others wait for initialization to complete.
     * Zero overhead after first call.
     * 
     * THIS IS THE SIMPLEST AND MOST EFFICIENT SOLUTION.
     */
    static Singleton& getInstance2() {
        static Singleton instance; // guaranteed thread-safe in C++11!
        return instance;
    }
    
    void doWork() { std::cout << "Working\n"; }
};

std::unique_ptr<Singleton> Singleton::instance_;
std::once_flag Singleton::onceFlag_;

// QUESTION 3: Two threads print alternately (1 thread prints 1-3-5,
//             other prints 2-4-6)
// ===================================================================
class AlternatePrinter {
    std::mutex mutex_;
    std::condition_variable cv_;
    int current_ = 1;
    const int max_;
    
public:
    explicit AlternatePrinter(int max) : max_(max) {}
    
    // Thread A: prints odd numbers
    void printOdd() {
        while (true) {
            std::unique_lock<std::mutex> lock(mutex_);
            cv_.wait(lock, [this]() { 
                return current_ > max_ || current_ % 2 == 1; 
            });
            if (current_ > max_) return;
            std::cout << current_ << " ";
            ++current_;
            cv_.notify_all();
        }
    }
    
    // Thread B: prints even numbers
    void printEven() {
        while (true) {
            std::unique_lock<std::mutex> lock(mutex_);
            cv_.wait(lock, [this]() { 
                return current_ > max_ || current_ % 2 == 0; 
            });
            if (current_ > max_) return;
            std::cout << current_ << " ";
            ++current_;
            cv_.notify_all();
        }
    }
};

void alternate_print_demo() {
    AlternatePrinter printer(20);
    
    std::thread odd([&printer]() { printer.printOdd(); });
    std::thread even([&printer]() { printer.printEven(); });
    
    odd.join();
    even.join();
    std::cout << "\n";
}

// QUESTION 4: Design a rate limiter
// ===================================================================
/*
 * TOKEN BUCKET ALGORITHM:
 * - Bucket has capacity N tokens
 * - Tokens added at rate R per second
 * - Each request consumes 1 token
 * - If no tokens: request is rejected or waits
 * 
 * ANALOGY: A restaurant kitchen token system
 * - Chef produces 5 orders/minute (refill rate)
 * - Kitchen can buffer 20 orders (bucket size)
 * - Each customer takes 1 order (consume 1 token)
 */

class RateLimiter {
    mutable std::mutex mutex_;
    double tokens_;          // current tokens
    const double maxTokens_; // bucket capacity
    const double refillRate_;// tokens per second
    std::chrono::steady_clock::time_point lastRefill_;
    
    void refill() {
        // Must be called under lock
        auto now = std::chrono::steady_clock::now();
        double elapsed = std::chrono::duration<double>(now - lastRefill_).count();
        
        tokens_ = std::min(maxTokens_, tokens_ + elapsed * refillRate_);
        lastRefill_ = now;
    }
    
public:
    RateLimiter(double maxTokens, double refillRatePerSecond)
        : tokens_(maxTokens)
        , maxTokens_(maxTokens)
        , refillRate_(refillRatePerSecond)
        , lastRefill_(std::chrono::steady_clock::now()) {}
    
    // Try to consume tokens (non-blocking)
    bool tryConsume(double tokens = 1.0) {
        std::lock_guard<std::mutex> lock(mutex_);
        refill();
        
        if (tokens_ >= tokens) {
            tokens_ -= tokens;
            return true;
        }
        return false;
    }
    
    // Block until tokens available
    void consume(double tokens = 1.0) {
        while (true) {
            {
                std::lock_guard<std::mutex> lock(mutex_);
                refill();
                if (tokens_ >= tokens) {
                    tokens_ -= tokens;
                    return;
                }
            }
            // Calculate wait time
            double needed = tokens - tokens_;
            double waitSeconds = needed / refillRate_;
            std::this_thread::sleep_for(
                std::chrono::duration<double>(waitSeconds * 0.5)
            );
        }
    }
    
    double available() const {
        std::lock_guard<std::mutex> lock(mutex_);
        const_cast<RateLimiter*>(this)->refill();
        return tokens_;
    }
};

// QUESTION 6: Parallel Map (apply function to array in parallel)
// ===================================================================
template<typename Iterator, typename Function>
void parallelMap(Iterator begin, Iterator end, Function fn, int numThreads = -1) {
    if (numThreads < 0) {
        numThreads = std::thread::hardware_concurrency();
    }
    
    size_t total = std::distance(begin, end);
    if (total == 0) return;
    
    size_t chunkSize = (total + numThreads - 1) / numThreads;
    
    std::vector<std::thread> threads;
    std::latch done(numThreads);
    
    auto it = begin;
    for (int t = 0; t < numThreads && it != end; t++) {
        auto chunkEnd = it;
        std::advance(chunkEnd, std::min(chunkSize, (size_t)std::distance(it, end)));
        
        threads.emplace_back([it, chunkEnd, &fn, &done]() {
            for (auto cur = it; cur != chunkEnd; ++cur) {
                fn(*cur);
            }
            done.count_down();
        });
        
        it = chunkEnd;
    }
    
    done.wait();
    for (auto& t : threads) t.join();
}

// QUESTION 7: Design a publish-subscribe system
// ===================================================================
#include <functional>
#include <unordered_map>
#include <unordered_set>

class EventBus {
    using Subscriber = std::function<void(const std::string&)>;
    using SubscriberID = int;
    
    struct Topic {
        std::unordered_map<SubscriberID, Subscriber> subscribers;
    };
    
    std::unordered_map<std::string, Topic> topics_;
    mutable std::shared_mutex rwMutex_;
    std::atomic<SubscriberID> nextId_{0};
    
    // Async dispatch
    BoundedStreamQueue<std::pair<std::string, std::string>> eventQueue_{10000};
    std::thread dispatchThread_;
    
public:
    EventBus() {
        // Background dispatch thread
        dispatchThread_ = std::thread([this]() {
            while (auto event = eventQueue_.pop()) {
                auto& [topic, data] = *event;
                
                // Get subscribers (read lock)
                std::vector<Subscriber> subs;
                {
                    std::shared_lock lock(rwMutex_);
                    auto it = topics_.find(topic);
                    if (it != topics_.end()) {
                        for (auto& [id, fn] : it->second.subscribers) {
                            subs.push_back(fn);
                        }
                    }
                }
                
                // Call subscribers (without holding lock!)
                for (auto& sub : subs) {
                    try {
                        sub(data);
                    } catch (const std::exception& e) {
                        std::cerr << "Subscriber threw: " << e.what() << "\n";
                    }
                }
            }
        });
    }
    
    // Subscribe to a topic, returns subscription ID
    SubscriberID subscribe(const std::string& topic, Subscriber fn) {
        std::unique_lock lock(rwMutex_);
        SubscriberID id = nextId_++;
        topics_[topic].subscribers[id] = std::move(fn);
        return id;
    }
    
    // Unsubscribe
    void unsubscribe(const std::string& topic, SubscriberID id) {
        std::unique_lock lock(rwMutex_);
        auto it = topics_.find(topic);
        if (it != topics_.end()) {
            it->second.subscribers.erase(id);
        }
    }
    
    // Publish event (async - returns immediately)
    void publish(const std::string& topic, std::string data) {
        eventQueue_.submit({topic, std::move(data)});
    }
    
    // Publish synchronously (wait for all subscribers to process)
    void publishSync(const std::string& topic, const std::string& data) {
        std::vector<Subscriber> subs;
        {
            std::shared_lock lock(rwMutex_);
            auto it = topics_.find(topic);
            if (it != topics_.end()) {
                for (auto& [id, fn] : it->second.subscribers) {
                    subs.push_back(fn);
                }
            }
        }
        for (auto& sub : subs) sub(data);
    }
    
    ~EventBus() {
        eventQueue_.close();
        if (dispatchThread_.joinable()) dispatchThread_.join();
    }
};
