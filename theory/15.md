#include <mutex>
#include <condition_variable>
#include <queue>
#include <thread>
#include <iostream>

// ===================================================================
// HOW CONDITION VARIABLES WORK INTERNALLY
// ===================================================================
/*
 * A condition variable is associated with:
 * 1. A mutex (for protecting the condition check)
 * 2. A wait queue (OS-level list of sleeping threads)
 * 
 * WAIT SEQUENCE (thread wants to wait):
 * ┌──────────────────────────────────────────────────────────────┐
 * │ 1. Acquire mutex (or already held)                          │
 * │ 2. Check condition (e.g., is queue empty?)                  │
 * │ 3. If condition met → continue (don't wait)                 │
 * │ 4. If NOT met:                                              │
 * │    a. ATOMICALLY: release mutex + add to wait queue         │
 * │       (these two must be atomic to avoid lost notifications) │
 * │    b. Thread goes to sleep (OS removes from CPU scheduler)  │
 * │    c. (Another thread notifies...)                          │
 * │    d. Thread is woken up                                    │
 * │    e. Re-acquire mutex                                      │
 * │    f. Re-check condition (spurious wakeup protection!)      │
 * │    g. If still not met → go back to step 4                  │
 * │    h. If met → continue                                     │
 * └──────────────────────────────────────────────────────────────┘
 * 
 * NOTIFY SEQUENCE (thread wants to wake others):
 * ┌──────────────────────────────────────────────────────────────┐
 * │ notify_one(): Wake ONE thread from wait queue               │
 * │ notify_all(): Wake ALL threads from wait queue              │
 * └──────────────────────────────────────────────────────────────┘
 * 
 * WHY SPURIOUS WAKEUPS?
 * On some OS/hardware, a waiting thread can wake up even
 * when not notified. This is NOT a bug — it's by design
 * to allow the OS to do housekeeping.
 * SOLUTION: ALWAYS check condition after waking!
 * ALWAYS use the predicate version of wait()!
 */

// ===================================================================
// PRODUCER-CONSUMER — COMPLETE IMPLEMENTATION
// ===================================================================

template<typename T>
class ConditionVariableQueue {
    std::queue<T> queue_;
    std::mutex mutex_;
    std::condition_variable cv_not_empty_; // signal: "item added"
    std::condition_variable cv_not_full_;  // signal: "space available"
    size_t maxCapacity_;
    bool closed_ = false;
    
public:
    explicit ConditionVariableQueue(size_t capacity) 
        : maxCapacity_(capacity) {}
    
    // ===== PRODUCER: Add item =====
    bool push(T item) {
        std::unique_lock<std::mutex> lock(mutex_);
        
        /*
         * WHY wait() needs unique_lock and NOT lock_guard?
         * wait() must RELEASE the mutex while sleeping.
         * lock_guard doesn't support manual unlock.
         * unique_lock does!
         * 
         * WHAT HAPPENS INTERNALLY:
         * wait(lock, pred) is equivalent to:
         *   while (!pred()) {
         *     lock.unlock();     // release mutex
         *     // sleep until notified
         *     // OS wakes us up
         *     lock.lock();       // reacquire mutex
         *   }
         * But the unlock+sleep is ATOMIC to prevent lost notifications
         */
        cv_not_full_.wait(lock, [this]() {
            return queue_.size() < maxCapacity_ || closed_;
        });
        
        if (closed_) return false;
        
        queue_.push(std::move(item));
        
        // Notify ONE consumer (not all — no need to wake everyone)
        cv_not_empty_.notify_one();
        
        return true;
        // lock released in destructor
    }
    
    // ===== CONSUMER: Remove item =====
    std::optional<T> pop() {
        std::unique_lock<std::mutex> lock(mutex_);
        
        cv_not_empty_.wait(lock, [this]() {
            return !queue_.empty() || closed_;
        });
        
        if (queue_.empty()) return std::nullopt; // closed + empty
        
        T item = std::move(queue_.front());
        queue_.pop();
        
        cv_not_full_.notify_one(); // signal producer: space available
        
        return item;
    }
    
    // ===== CONSUMER WITH TIMEOUT =====
    std::optional<T> pop_with_timeout(std::chrono::milliseconds timeout) {
        std::unique_lock<std::mutex> lock(mutex_);
        
        /*
         * wait_for returns:
         * - true: condition became true (got item)
         * - false: timed out
         */
        bool success = cv_not_empty_.wait_for(lock, timeout, [this]() {
            return !queue_.empty() || closed_;
        });
        
        if (!success || queue_.empty()) return std::nullopt;
        
        T item = std::move(queue_.front());
        queue_.pop();
        cv_not_full_.notify_one();
        return item;
    }
    
    // ===== SHUTDOWN: Wake everyone up to exit =====
    /*
     * WHY SHUTDOWN IS TRICKY:
     * Consumers might be sleeping in wait().
     * If producers finish and we don't signal, consumers sleep forever!
     * 
     * Solution: set closed_ flag, notify_all() to wake all waiters.
     * Waiters check if closed_ is true and exit gracefully.
     */
    void close() {
        {
            std::lock_guard<std::mutex> lock(mutex_);
            closed_ = true;
        }
        // Wake ALL threads (both producers and consumers might be waiting)
        cv_not_empty_.notify_all();
        cv_not_full_.notify_all();
    }
    
    bool empty() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return queue_.empty();
    }
    
    size_t size() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return queue_.size();
    }
};

// ===================================================================
// DEMONSTRATION: Multiple Producers and Consumers
// ===================================================================

void producer_consumer_full_demo() {
    ConditionVariableQueue<int> queue(5); // capacity of 5
    
    const int NUM_PRODUCERS = 3;
    const int NUM_CONSUMERS = 2;
    const int ITEMS_PER_PRODUCER = 10;
    
    std::vector<std::thread> producers;
    std::vector<std::thread> consumers;
    
    std::atomic<int> totalProduced{0};
    std::atomic<int> totalConsumed{0};
    
    // Start producers
    for (int p = 0; p < NUM_PRODUCERS; p++) {
        producers.emplace_back([&, p]() {
            for (int i = 0; i < ITEMS_PER_PRODUCER; i++) {
                int item = p * 100 + i;
                if (queue.push(item)) {
                    totalProduced++;
                    std::cout << "P" << p << " → " << item << "\n";
                }
            }
            std::cout << "Producer " << p << " done\n";
        });
    }
    
    // Start consumers
    for (int c = 0; c < NUM_CONSUMERS; c++) {
        consumers.emplace_back([&, c]() {
            while (true) {
                auto item = queue.pop();
                if (!item.has_value()) {
                    std::cout << "Consumer " << c << " exiting\n";
                    break;
                }
                totalConsumed++;
                std::cout << "C" << c << " ← " << *item << "\n";
                std::this_thread::sleep_for(std::chrono::milliseconds(50));
            }
        });
    }
    
    // Wait for producers to finish
    for (auto& p : producers) p.join();
    
    // Signal consumers: no more data coming
    queue.close();
    
    // Wait for consumers to finish
    for (auto& c : consumers) c.join();
    
    std::cout << "Produced: " << totalProduced 
              << " Consumed: " << totalConsumed << "\n";
}

// COMMON BUG: LOST NOTIFICATION
// ===================================================================
/*
 * SCENARIO:
 * 
 * Consumer:                    Producer:
 *   Check: queue empty? YES      (hasn't added yet)
 *   (about to wait...)
 *                                Add item
 *                                notify_one() ← LOST! no one waiting yet
 *   wait() ← sleeps forever
 * 
 * SOLUTION: The predicate check and the wait are done atomically
 * under the mutex. Producer must hold the mutex when modifying
 * queue AND when calling notify.
 * 
 * ACTUALLY: You can notify OUTSIDE the lock:
 * 
 * {
 *   lock_guard lock(mtx);
 *   queue.push(item);
 * } // ← unlock here
 * cv.notify_one(); // ← notify after unlock
 * 
 * This can be slightly faster (woken thread doesn't have to wait
 * for notifier to release lock) but is always safe.
 * The wait's predicate check is under the mutex, so no race.
 */
