PART 13: PATTERNS AND PUTTING IT ALL TOGETHER
13.1 Monitor Pattern — Encapsulated Synchronization

```cpp
/*
 * MONITOR PATTERN:
 * Encapsulate: data + mutex + condition variables together.
 * All access to data goes through monitor's synchronized methods.
 * 
 * This is the CLEANEST way to write thread-safe classes.
 * Used everywhere in Java's synchronized methods.
 * 
 * PRINCIPLES:
 * 1. Mutex and data are ALWAYS together (in same class)
 * 2. All public methods that touch data acquire the lock first
 * 3. Condition variables live inside the monitor
 * 4. Internal implementation methods don't lock (assume called under lock)
 */

#include <mutex>
#include <condition_variable>
#include <queue>
#include <optional>

// ===== A COMPLETE MONITOR: Bounded Work Queue =====
template<typename Task>
class WorkQueueMonitor {
    // =========================================================
    // PRIVATE: Data + Synchronization (all together)
    // =========================================================
    std::queue<Task> tasks_;
    mutable std::mutex mutex_;
    std::condition_variable notEmpty_;
    std::condition_variable notFull_;
    
    const size_t maxSize_;
    bool stopped_ = false;
    size_t activeWorkers_ = 0;
    
    // =========================================================
    // PRIVATE: Internal methods (assume lock is held!)
    // =========================================================
    bool isEmpty_() const { return tasks_.empty(); }
    bool isFull_() const { return tasks_.size() >= maxSize_; }
    bool isDone_() const { return stopped_ && tasks_.empty() && activeWorkers_ == 0; }
    
public:
    // =========================================================
    // PUBLIC: All synchronized
    // =========================================================
    
    explicit WorkQueueMonitor(size_t maxSize) : maxSize_(maxSize) {}
    
    // Producer: submit work
    bool submit(Task task) {
        std::unique_lock<std::mutex> lock(mutex_);
        
        notFull_.wait(lock, [this]() { return !isFull_() || stopped_; });
        
        if (stopped_) return false;
        
        tasks_.push(std::move(task));
        notEmpty_.notify_one();
        return true;
    }
    
    // Consumer: get work (blocking)
    std::optional<Task> getWork() {
        std::unique_lock<std::mutex> lock(mutex_);
        
        ++activeWorkers_;
        
        // Wait for work OR shutdown
        notEmpty_.wait(lock, [this]() { return !isEmpty_() || stopped_; });
        
        if (isEmpty_()) {
            --activeWorkers_;
            return std::nullopt; // shutting down
        }
        
        Task task = std::move(tasks_.front());
        tasks_.pop();
        
        notFull_.notify_one();
        return task;
    }
    
    // Consumer: signal work completed
    void workCompleted() {
        std::unique_lock<std::mutex> lock(mutex_);
        --activeWorkers_;
        
        // If last worker AND stopped AND empty: everything done
        if (isDone_()) {
            notEmpty_.notify_all();
        }
    }
    
    // Control: initiate shutdown
    void shutdown(bool drainQueue = true) {
        {
            std::lock_guard<std::mutex> lock(mutex_);
            stopped_ = true;
            if (!drainQueue) {
                while (!tasks_.empty()) tasks_.pop();
            }
        }
        notEmpty_.notify_all();
        notFull_.notify_all();
    }
    
    // Wait until all work is processed
    void waitUntilDone() {
        std::unique_lock<std::mutex> lock(mutex_);
        notEmpty_.wait(lock, [this]() { return isDone_(); });
    }
    
    // Inspection
    size_t queueSize() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return tasks_.size();
    }
    
    size_t activeWorkerCount() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return activeWorkers_;
    }
    
    bool isShutdown() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return stopped_;
    }
};
```
