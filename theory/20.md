8.2 Implementing Reader-Writer Lock from Scratch

```cpp
#include <mutex>
#include <condition_variable>
#include <shared_mutex>
#include <thread>
#include <vector>
#include <iostream>
#include <chrono>

// ===================================================================
// READER-WRITER LOCK FROM SCRATCH
// (This is a classic Google interview question)
// ===================================================================

class ReaderWriterLock {
    /*
     * STATE WE NEED TO TRACK:
     * - How many readers are currently reading?
     * - Is a writer currently writing?
     * - How many writers are waiting? (for writer priority)
     * 
     * INVARIANTS:
     * 1. Multiple readers can read simultaneously (if no writer active)
     * 2. At most ONE writer can write at a time
     * 3. No readers allowed while writer is writing
     * 4. No new readers allowed while writers are waiting (writer priority)
     */
    
    std::mutex mtx_;
    std::condition_variable cv_readers_;  // signal: "readers can proceed"
    std::condition_variable cv_writers_;  // signal: "writer can proceed"
    
    int activeReaders_ = 0;
    bool activeWriter_ = false;
    int waitingWriters_ = 0;
    
public:
    // ===== READER LOCK =====
    void lockRead() {
        std::unique_lock<std::mutex> lock(mtx_);
        
        /*
         * Reader must wait if:
         * 1. A writer is currently writing, OR
         * 2. Writers are waiting (writer priority — prevents starvation)
         * 
         * Without (2), many readers could keep coming and starve writers.
         */
        cv_readers_.wait(lock, [this]() {
            return !activeWriter_ && waitingWriters_ == 0;
        });
        
        ++activeReaders_;
        // Note: we don't unlock here — lock goes out of scope
        // But mutex is released when this function returns
        // (other threads can then acquire the mutex to check state)
    }
    
    void unlockRead() {
        std::unique_lock<std::mutex> lock(mtx_);
        --activeReaders_;
        
        if (activeReaders_ == 0 && waitingWriters_ > 0) {
            // Last reader done AND writers waiting → wake a writer
            cv_writers_.notify_one();
        }
    }
    
    // ===== WRITER LOCK =====
    void lockWrite() {
        std::unique_lock<std::mutex> lock(mtx_);
        
        ++waitingWriters_; // signal our intent (blocks new readers)
        
        /*
         * Writer must wait until:
         * 1. No active readers (all reads done)
         * 2. No active writer (last writer done)
         */
        cv_writers_.wait(lock, [this]() {
            return activeReaders_ == 0 && !activeWriter_;
        });
        
        --waitingWriters_;
        activeWriter_ = true;
    }
    
    void unlockWrite() {
        std::unique_lock<std::mutex> lock(mtx_);
        activeWriter_ = false;
        
        if (waitingWriters_ > 0) {
            // Prefer waking a writer (writer priority)
            cv_writers_.notify_one();
        } else {
            // No writers waiting, wake all readers
            cv_readers_.notify_all();
        }
    }
    
    // ===== RAII WRAPPERS =====
    class ReadLock {
        ReaderWriterLock& rwl_;
    public:
        explicit ReadLock(ReaderWriterLock& rwl) : rwl_(rwl) { rwl_.lockRead(); }
        ~ReadLock() { rwl_.unlockRead(); }
        ReadLock(const ReadLock&) = delete;
    };
    
    class WriteLock {
        ReaderWriterLock& rwl_;
    public:
        explicit WriteLock(ReaderWriterLock& rwl) : rwl_(rwl) { rwl_.lockWrite(); }
        ~WriteLock() { rwl_.unlockWrite(); }
        WriteLock(const WriteLock&) = delete;
    };
};

// ===================================================================
// FAIR READER-WRITER LOCK (No Starvation for Either Side)
// ===================================================================
/*
 * Problem with writer-priority lock: if writers keep coming,
 * readers might wait forever (reader starvation).
 * 
 * Solution: Use a ticket/queue system — requests served in arrival order.
 * Reader after a writer → waits for that writer.
 * Writer after readers → waits for those readers.
 * 
 * INTERNAL MECHANISM:
 * Each lock request gets a ticket number.
 * When a "turn" completes, the next ticket(s) are served.
 * Consecutive reader tickets can be served together.
 */

class FairRWLock {
    std::mutex mtx_;
    std::condition_variable cv_;
    
    int activeReaders_ = 0;
    bool activeWriter_ = false;
    
    // Ticket system
    struct Waiter {
        bool isWriter;
        bool ready = false;
    };
    std::deque<Waiter*> waiters_;
    
public:
    void lockRead() {
        std::unique_lock<std::mutex> lock(mtx_);
        
        // Can proceed immediately if: no writer active AND
        // no one in queue (fair: don't cut in line)
        if (!activeWriter_ && waiters_.empty()) {
            ++activeReaders_;
            return;
        }
        
        Waiter w{false, false};
        waiters_.push_back(&w);
        
        cv_.wait(lock, [&w]() { return w.ready; });
        ++activeReaders_;
    }
    
    void unlockRead() {
        std::unique_lock<std::mutex> lock(mtx_);
        --activeReaders_;
        
        if (activeReaders_ == 0) {
            wakeNext(lock);
        }
    }
    
    void lockWrite() {
        std::unique_lock<std::mutex> lock(mtx_);
        
        if (!activeWriter_ && activeReaders_ == 0 && waiters_.empty()) {
            activeWriter_ = true;
            return;
        }
        
        Waiter w{true, false};
        waiters_.push_back(&w);
        
        cv_.wait(lock, [&w]() { return w.ready; });
        activeWriter_ = true;
    }
    
    void unlockWrite() {
        std::unique_lock<std::mutex> lock(mtx_);
        activeWriter_ = false;
        wakeNext(lock);
    }

private:
    void wakeNext(std::unique_lock<std::mutex>&) {
        if (waiters_.empty()) return;
        
        if (waiters_.front()->isWriter) {
            // Next in line is a writer: wake just them
            waiters_.front()->ready = true;
            waiters_.pop_front();
        } else {
            // Next in line is a reader: wake ALL consecutive readers
            while (!waiters_.empty() && !waiters_.front()->isWriter) {
                waiters_.front()->ready = true;
                waiters_.pop_front();
            }
        }
        cv_.notify_all(); // wake appropriate waiters
    }
};

// ===================================================================
// std::shared_mutex — PRODUCTION USE
// ===================================================================
/*
 * C++ standard library provides std::shared_mutex (C++17)
 * Use this in production instead of custom RW locks.
 * 
 * std::shared_mutex:
 * - Pair with std::shared_lock for readers (multiple at once)
 * - Pair with std::unique_lock for writers (exclusive)
 * 
 * std::shared_timed_mutex:
 * - Same but supports try_lock_for() and try_lock_until()
 */

class ProductionThreadSafeMap {
    mutable std::shared_mutex rwMutex_;
    std::map<std::string, int> data_;
    
public:
    // READER: Can run concurrently with other readers
    int get(const std::string& key) const {
        std::shared_lock lock(rwMutex_); // C++17 CTAD
        auto it = data_.find(key);
        return (it != data_.end()) ? it->second : -1;
    }
    
    // WRITER: Exclusive access
    void set(const std::string& key, int value) {
        std::unique_lock lock(rwMutex_);
        data_[key] = value;
    }
    
    // READ-MODIFY-WRITE: Need to handle carefully
    bool increment(const std::string& key) {
        // OPTION A: Upgrade pattern (release read, acquire write)
        {
            std::shared_lock readLock(rwMutex_);
            if (data_.find(key) == data_.end()) return false;
        } // release read lock
        
        {
            std::unique_lock writeLock(rwMutex_);
            // MUST re-check! Another thread might have deleted it
            auto it = data_.find(key);
            if (it == data_.end()) return false;
            ++(it->second);
            return true;
        }
    }
    
    // OPTION B: Just use write lock throughout (simpler, safer)
    bool safe_increment(const std::string& key) {
        std::unique_lock writeLock(rwMutex_);
        auto it = data_.find(key);
        if (it == data_.end()) return false;
        ++(it->second);
        return true;
    }
};

// PERFORMANCE TEST: mutex vs shared_mutex
// ===================================================================
void performance_comparison() {
    const int NUM_READERS = 10;
    const int NUM_READS = 100000;
    
    std::string sharedData = "hello";
    
    // Test 1: Regular mutex (readers serialized)
    {
        std::mutex mtx;
        auto start = std::chrono::high_resolution_clock::now();
        
        std::vector<std::thread> threads;
        for (int i = 0; i < NUM_READERS; i++) {
            threads.emplace_back([&]() {
                for (int j = 0; j < NUM_READS; j++) {
                    std::lock_guard<std::mutex> lock(mtx);
                    volatile auto x = sharedData.length(); // force read
                    (void)x;
                }
            });
        }
        for (auto& t : threads) t.join();
        
        auto duration = std::chrono::high_resolution_clock::now() - start;
        std::cout << "Mutex: " 
                  << std::chrono::duration_cast<std::chrono::milliseconds>(duration).count()
                  << "ms\n";
    }
    
    // Test 2: shared_mutex (readers concurrent)
    {
        std::shared_mutex rwMtx;
        auto start = std::chrono::high_resolution_clock::now();
        
        std::vector<std::thread> threads;
        for (int i = 0; i < NUM_READERS; i++) {
            threads.emplace_back([&]() {
                for (int j = 0; j < NUM_READS; j++) {
                    std::shared_lock<std::shared_mutex> lock(rwMtx);
                    volatile auto x = sharedData.length();
                    (void)x;
                }
            });
        }
        for (auto& t : threads) t.join();
        
        auto duration = std::chrono::high_resolution_clock::now() - start;
        std::cout << "shared_mutex: " 
                  << std::chrono::duration_cast<std::chrono::milliseconds>(duration).count()
                  << "ms\n";
    }
    
    // shared_mutex typically 2-10x faster for read-heavy workloads
}
```
