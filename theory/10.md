4.2 Every Type of Lock — Why Each Exists

```cpp
#include <mutex>
#include <shared_mutex>
#include <thread>
#include <iostream>
#include <vector>
#include <chrono>

// ===================================================================
// LOCK WRAPPER 1: std::lock_guard
// ===================================================================
/*
 * PROBLEM IT SOLVES:
 * 
 * void badFunction(std::mutex& m) {
 *     m.lock();
 *     doWork();          // if this throws → m.unlock() NEVER called!
 *     m.unlock();        // resource LEAKED, other threads DEADLOCK
 * }
 * 
 * SOLUTION: RAII — acquire in constructor, release in destructor
 * Even if exception is thrown, destructor is called = unlock guaranteed
 * 
 * WHEN TO USE: Simple cases, don't need to unlock manually
 * LIMITATION: Can't unlock before scope ends
 */

std::mutex counterMtx;
int counter = 0;

void with_lock_guard() {
    std::lock_guard<std::mutex> guard(counterMtx);
    // ↑ mutex locked here
    
    counter++;
    // even if exception here, guard's destructor unlocks!
    
} // ← guard destructor called: mutex unlocked automatically

// C++17 simplification (CTAD):
void with_lock_guard_cpp17() {
    std::lock_guard guard(counterMtx); // no need to specify type
    counter++;
}

// ===================================================================
// LOCK WRAPPER 2: std::unique_lock
// ===================================================================
/*
 * PROBLEM IT SOLVES:
 * lock_guard is inflexible — you can't:
 * - Unlock before scope ends (to reduce contention)
 * - Try to lock without blocking
 * - Lock with a timeout
 * - Transfer lock ownership
 * - Use with condition_variable (REQUIRES unique_lock)
 * 
 * unique_lock gives you ALL of this flexibility.
 * 
 * INTERNAL STATE:
 * unique_lock stores:
 *   - Pointer to the mutex
 *   - bool: whether it currently holds the lock
 * 
 * COST: Slightly more overhead than lock_guard (extra bool check)
 */

std::timed_mutex timedMtx;

void with_unique_lock_examples() {
    
    // === IMMEDIATE LOCK (same as lock_guard) ===
    {
        std::unique_lock<std::mutex> lock(counterMtx);
        counter++;
    } // unlocks here
    
    // === DEFERRED LOCK ===
    // Useful when you don't want to lock right away
    {
        std::unique_lock<std::mutex> lock(counterMtx, std::defer_lock);
        // mutex NOT locked yet!
        
        // do non-critical work...
        int temp = 42;
        
        lock.lock(); // lock when ready
        counter += temp;
        
        lock.unlock(); // unlock early if needed
        
        // more non-critical work...
    } // destructor checks if locked → won't double-unlock
    
    // === TRY LOCK (non-blocking) ===
    {
        std::unique_lock<std::mutex> lock(counterMtx, std::try_to_lock);
        if (lock.owns_lock()) {
            counter++; // got the lock, do work
        } else {
            // couldn't get lock, do alternative work
            std::cout << "Lock busy, skipping\n";
        }
    }
    
    // === TIMED LOCK ===
    {
        std::unique_lock<std::timed_mutex> lock(
            timedMtx, 
            std::chrono::milliseconds(100) // wait at most 100ms
        );
        if (lock.owns_lock()) {
            std::cout << "Got timed lock\n";
        } else {
            std::cout << "Timed out waiting for lock\n";
        }
    }
    
    // === ADOPT LOCK (I already locked it manually) ===
    counterMtx.lock(); // manually locked
    {
        // Tell unique_lock "you already own this lock"
        std::unique_lock<std::mutex> lock(counterMtx, std::adopt_lock);
        counter++;
    } // unique_lock unlocks in destructor
    // Note: don't call counterMtx.unlock() separately!
    
    // === TRANSFER OWNERSHIP ===
    auto getLock = [&]() {
        return std::unique_lock<std::mutex>(counterMtx);
    };
    auto lock = getLock(); // can move unique_lock
    counter++;
    // lock_guard CANNOT be moved or returned like this
}

// ===================================================================
// LOCK WRAPPER 3: std::scoped_lock (C++17)
// ===================================================================
/*
 * PROBLEM IT SOLVES: Locking MULTIPLE mutexes safely
 * 
 * Classic deadlock scenario:
 * 
 * Thread 1: lock(A), lock(B) → waits for B while holding A
 * Thread 2: lock(B), lock(A) → waits for A while holding B
 * DEADLOCK! Both waiting forever.
 * 
 * scoped_lock uses a deadlock-avoidance algorithm internally:
 * It tries to lock all mutexes at once, backing off if any fail.
 * Uses std::lock() internally which implements deadlock-free
 * multi-lock acquisition.
 * 
 * ALGORITHM INTERNALLY (simplified):
 * 1. Try to lock mutex 1
 * 2. If got it, try to lock mutex 2
 * 3. If mutex 2 fails, RELEASE mutex 1
 * 4. Try again (this prevents deadlock by backing off)
 * 
 * This is called "try-and-back-off" or uses the "backing off" strategy
 */

struct Account {
    std::mutex mtx;
    double balance;
    int id;
    
    Account(int id, double bal) : id(id), balance(bal) {}
};

void safeTransfer(Account& from, Account& to, double amount) {
    // Lock BOTH simultaneously — deadlock impossible!
    std::scoped_lock lock(from.mtx, to.mtx);
    
    if (from.balance >= amount) {
        from.balance -= amount;
        to.balance += amount;
        std::cout << "Transferred " << amount 
                  << " from " << from.id 
                  << " to " << to.id << "\n";
    }
} // both mutexes unlocked here

// ===================================================================
// LOCK WRAPPER 4: std::shared_lock
// ===================================================================
/*
 * PROBLEM IT SOLVES:
 * Regular mutex allows only ONE thread at a time — even for READING!
 * This is unnecessarily restrictive.
 * 
 * If multiple threads are only READING shared data, they can safely
 * do so simultaneously — reads don't interfere with each other!
 * 
 * BUT: A write must be exclusive — no one else reading or writing.
 * 
 * shared_lock is the "reader lock" for std::shared_mutex:
 * - Multiple threads can hold shared_lock simultaneously
 * - If ANY thread holds shared_lock, exclusive lock (unique_lock) blocks
 * - If any thread holds unique_lock, all shared_locks block
 * 
 * INTERNALLY:
 * shared_mutex maintains:
 *   - Count of current readers
 *   - Whether a writer is active or waiting
 *   - Wait queues for readers and writers
 */

#include <shared_mutex>

class ConfigManager {
    mutable std::shared_mutex rwMutex; // 'mutable' allows locking in const methods
    std::map<std::string, std::string> config;
    
public:
    // READER: Multiple can run simultaneously
    std::string get(const std::string& key) const {
        std::shared_lock<std::shared_mutex> readLock(rwMutex);
        // ↑ Multiple threads can be here simultaneously!
        
        auto it = config.find(key);
        return (it != config.end()) ? it->second : "";
    }
    
    // WRITER: Exclusive access
    void set(const std::string& key, const std::string& value) {
        std::unique_lock<std::shared_mutex> writeLock(rwMutex);
        // ↑ Only ONE thread can be here, no readers allowed!
        
        config[key] = value;
    }
};

/*
 * PERFORMANCE COMPARISON:
 * 
 * With std::mutex (everyone waits):
 * T1 read ─────►|
 *               T2 read ─────►|
 *                              T3 read ─────►|
 * Total time: 3x read_time
 * 
 * With std::shared_mutex (readers concurrent):
 * T1 read ─────►|
 * T2 read ─────►| (concurrent!)
 * T3 read ─────►| (concurrent!)
 * Total time: 1x read_time
 * 
 * 3x speedup for read-heavy workloads!
 */
```
